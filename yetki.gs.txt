// ── SHEET'TEN YETKİ OKU ────────────────────────────────────────
function api_yetkiRolleriniGetir() {
  try {
    var ss = SpreadsheetApp.openById(CRM_SHEET_ID);
    var sh = ss.getSheetByName(YETKI_SHEET);

    // Sheet yoksa varsayılanları kullan
    if (!sh) {
      return { ok: true, roller: VARSAYILAN_YETKILER, tanimlar: IZIN_TANIMLARI, sistemRolleri: SISTEM_ROLLERI };
    }

    var data = sh.getDataRange().getValues();
    if (data.length < 2) {
      return { ok: true, roller: VARSAYILAN_YETKILER, tanimlar: IZIN_TANIMLARI, sistemRolleri: SISTEM_ROLLERI };
    }

    // Başlık satırından rol sıralamasını al (col 4+)
    var baslik = data[0];
    var rolSiralamasi = baslik.slice(3); // ["ADMIN","MUDUR","ISG",...]

    var roller = {};
    rolSiralamasi.forEach(function(r) { roller[r] = {}; });

    for (var i = 1; i < data.length; i++) {
      var izinKey = String(data[i][0] || "").trim();
      if (!izinKey) continue;
      rolSiralamasi.forEach(function(rol, idx) {
        var deger = data[i][3 + idx];
        roller[rol][izinKey] = (deger === true || String(deger).toUpperCase() === "TRUE");
      });
    }

    return { ok: true, roller: roller, tanimlar: IZIN_TANIMLARI, sistemRolleri: SISTEM_ROLLERI };
  } catch(e) {
    return { ok: false, msg: e.message, roller: VARSAYILAN_YETKILER, tanimlar: IZIN_TANIMLARI, sistemRolleri: SISTEM_ROLLERI };
  }
}

// ── YETKİLERİ KAYDET ──────────────────────────────────────────
function api_yetkiKaydet(p) {
  try {
    var email = String(p.email || "").toLowerCase().trim();
    var u = AUTH_requireKnownUser_(email);
    var rol = String(u.role || "").toUpperCase();
    if (rol !== "ADMIN") return { ok: false, msg: "Sadece ADMIN yetki değişikliği yapabilir." };

    var roller = p.roller; // { "MUDUR": { "dashboard": true, ... }, ... }
    if (!roller) return { ok: false, msg: "Rol verisi eksik." };

    var ss = SpreadsheetApp.openById(CRM_SHEET_ID);
    var sh = ss.getSheetByName(YETKI_SHEET);
    if (!sh) {
      sh = ss.insertSheet(YETKI_SHEET);
    }

    sh.clearContents();

    // Başlık satırı
    var baslik = ["İzin Anahtarı", "Grup", "Açıklama"].concat(SISTEM_ROLLERI);
    sh.appendRow(baslik);
    sh.getRange(1, 1, 1, baslik.length)
      .setFontWeight("bold")
      .setBackground("#1e293b")
      .setFontColor("white");

    // Her izin için satır yaz
    IZIN_TANIMLARI.forEach(function(tanim) {
      var satir = [tanim.key, tanim.grup, tanim.etiket];
      SISTEM_ROLLERI.forEach(function(rol) {
        var rolYetkiler = roller[rol] || VARSAYILAN_YETKILER[rol] || {};
        satir.push(rolYetkiler[tanim.key] === true);
      });
      sh.appendRow(satir);
    });

    // Sütun genişliklerini ayarla
    sh.setColumnWidth(1, 200);
    sh.setColumnWidth(2, 120);
    sh.setColumnWidth(3, 260);
    for (var c = 4; c <= 3 + SISTEM_ROLLERI.length; c++) {
      sh.setColumnWidth(c, 90);
    }
    sh.setFrozenRows(1);
    sh.setFrozenColumns(3);

    return { ok: true, msg: "Yetki tablosu kaydedildi." };
  } catch(e) {
    return { ok: false, msg: e.message };
  }
}

// ── KULLANICININ YETKİSİNİ AL ─────────────────────────────────
// Login sırasında veya izin kontrolünde kullanılır
function api_kullaniciYetkisiGetir(email) {
  try {
    var u = AUTH_getUserProfile_(email);
    if (!u) return { ok: false, msg: "Kullanıcı bulunamadı." };

    var rol = String(u.role || "PERSONEL").toUpperCase();

    // YETKI_ROLLERI sheet'i varsa oradan oku, yoksa varsayılan kullan
    var res = api_yetkiRolleriniGetir();
    var roller = res.roller || VARSAYILAN_YETKILER;
    var izinler = roller[rol] || roller["PERSONEL"] || {};

    // ADMIN her zaman her şeye erişebilir
    if (rol === "ADMIN") {
      izinler = VARSAYILAN_YETKILER["ADMIN"];
    }

    return { ok: true, email: u.email, rol: rol, adSoyad: u.adSoyad, izinler: izinler };
  } catch(e) {
    return { ok: false, msg: e.message };
  }
}

// ── YETKİ KONTROL YARDIMCISI (backend middleware) ─────────────
// Her api_ fonksiyonunun başında çağrılabilir
function YETKI_kontrol_(email, izinKey) {
  try {
    var res = api_kullaniciYetkisiGetir(email);
    if (!res.ok) throw new Error("Kullanıcı doğrulanamadı.");
    if (res.rol === "ADMIN") return true; // ADMIN her şeyi yapabilir
    if (res.izinler[izinKey] !== true) {
      throw new Error("Bu işlem için yetkiniz yok: " + izinKey);
    }
    return true;
  } catch(e) {
    throw e;
  }
}

// ── KULLANICI LİSTESİ İÇİN ────────────────────────────────────
function api_kullaniciListesiGetir(email) {
  try {
    YETKI_kontrol_(email, "kullanici_yonetimi");
    var ss = SpreadsheetApp.openById(CRM_SHEET_ID);
    var sh = ss.getSheetByName("Kullanicilar");
    if (!sh) return { ok: false, msg: "Kullanicilar sheet bulunamadı." };
    var data = sh.getDataRange().getValues();
    var liste = [];
    for (var i = 1; i < data.length; i++) {
      if (!data[i][0]) continue;
      liste.push({
        email:   String(data[i][0] || ""),
        pin:     "****",
        rol:     String(data[i][2] || "PERSONEL"),
        adSoyad: String(data[i][3] || ""),
        tel:     String(data[i][4] || ""),
        aktif:   String(data[i][6] || "AKTIF").toUpperCase() !== "PASİF"
      });
    }
    return { ok: true, liste: liste };
  } catch(e) {
    return { ok: false, msg: e.message };
  }
}

// ── KULLANICI ROLÜ GÜNCELLE ───────────────────────────────────
function api_kullaniciRoluGuncelle(p) {
  try {
    YETKI_kontrol_(p.adminEmail, "kullanici_yonetimi");
    var ss = SpreadsheetApp.openById(CRM_SHEET_ID);
    var sh = ss.getSheetByName("Kullanicilar");
    if (!sh) return { ok: false, msg: "Kullanicilar sheet bulunamadı." };
    var data = sh.getDataRange().getValues();
    var hedefEmail = String(p.hedefEmail || "").toLowerCase().trim();
    for (var i = 1; i < data.length; i++) {
      if (String(data[i][0] || "").toLowerCase().trim() === hedefEmail) {
        sh.getRange(i + 1, 3).setValue(String(p.yeniRol || "PERSONEL").toUpperCase());
        return { ok: true, msg: hedefEmail + " kullanıcısının rolü güncellendi: " + p.yeniRol };
      }
    }
    return { ok: false, msg: "Kullanıcı bulunamadı: " + hedefEmail };
  } catch(e) {
    return { ok: false, msg: e.message };
  }
}
