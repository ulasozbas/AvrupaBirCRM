function escapeHtml_(s) {
  return String(s || "").replace(/[&<>"']/g, function(m) {
  return { "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[m];
  });
}

  ];

  var container = document.getElementById("userCardsContainer");
  if (!container) return;

  var scoresList = [];

  users.forEach(function(u) {
