// RAPOR - KULLANICI PERFORMANS
// ==========================================
function api_raporKullaniciPerformans(opts) {
  opts = opts || {};
  var ss = SpreadsheetApp.openById(CRM_SHEET_ID);
  var sh = ss.getSheetByName("HAREKETLER");
  if (!sh) throw new Error("HAREKETLER sayfası bulunamadı!");
  var vals = sh.getDataRange().getDisplayValues();
  if (vals.length < 2) return { ok:true, rows:[], summary:{} };
  var head = vals[0];
  var idx = {};
  head.forEach(function(h,i){ idx[String(h||"").toLowerCase().replace(/\s+/g,"").trim()] = i; });
  var ixTarih=idx["tarih"], ixFirma=idx["firmaunvani"], ixUser=idx["kullaniciemail"], ixKanal=idx["kanal"], ixIslem=idx["islem"], ixAsama=idx["asama"];
  function s(x){ return String(x==null?"":x).trim(); }
  function parseTRDate_(t) {
    t = s(t); if (!t) return null;
    var parts = t.split(" ")[0].split(".");
    if (parts.length < 3) return null;
    return new Date(parts[2], parts[1]-1, parts[0]);
  }
  var d1 = opts.from ? parseTRDate_(opts.from) : null;
  var d2 = opts.to ? parseTRDate_(opts.to) : null;
  if (d2) d2.setHours(23,59,59,999);
  var fKanal = s(opts.kanal).toUpperCase(), fAsama = s(opts.asama).toUpperCase(), fUser = s(opts.kullaniciEmail).toLowerCase();
  var rows=[], byKanal={}, byAsama={}, sayIslem={TEKLIF:0,TANITIM:0,TOPLANTI:0}, pos=0, neg=0;
  for (var i=vals.length-1; i>=1; i--) {
    var r=vals[i];
    var rDate=parseTRDate_(r[ixTarih]);
    if (d1&&rDate&&rDate<d1) continue;
    if (d2&&rDate&&rDate>d2) continue;
    var kanal=s(r[ixKanal]).toUpperCase(), asama=s(r[ixAsama]).toUpperCase(), user=s(r[ixUser]).toLowerCase();
    if (fKanal&&kanal!==fKanal) continue;
    if (fAsama&&asama!==fAsama) continue;
    if (fUser&&user!==fUser) continue;
    byKanal[kanal]=(byKanal[kanal]||0)+1;
    byAsama[asama]=(byAsama[asama]||0)+1;
    var islemTipi=s(r[ixIslem]).toUpperCase();
    if(islemTipi==="TEKLIF") sayIslem.TEKLIF++;
    if(islemTipi==="TANITIM") sayIslem.TANITIM++;
    if(islemTipi==="TOPLANTI") sayIslem.TOPLANTI++;
    if(asama==="OLUMLU") pos++;
    if(asama==="OLUMSUZ") neg++;
    if(kanal === "SİSTEM" || kanal === "SISTEM") continue;
    rows.push({ tarih:String(r[ixTarih]||""), firma:r[ixFirma]||"-", user:user.split("@")[0], kanal:kanal, islem:islemTipi, asama:asama });
  }
  return { ok:true, summary:{ toplamAktivite:rows.length, kanalDagilim:byKanal, asamaDagilim:byAsama, sayilar:sayIslem, oran:{ olumlu:pos, olumsuz:neg, olumluOran:(pos+neg)?Math.round((pos/(pos+neg))*100):0 } }, rows:rows };
}

function api_raporTeklifler(p) {
  try {
    p = p || {};
    var ss = SpreadsheetApp.openById(CRM_SHEET_ID);
    var sh = ss.getSheetByName("TEKLIFLER") || ss.getSheetByName("TEKLIFLER");
    if (!sh || sh.getLastRow() < 2) return { ok:true, rows:[] };
    var vals = sh.getDataRange().getValues();
    var head = vals[0];
    function getIdx(names){ for(var n=0;n<names.length;n++){var i=head.indexOf(names[n]);if(i>-1)return i;} return -1; }
    var iNo=getIdx(["TeklifNo"]), iTarih=getIdx(["Tarih","OlusturmaTarihi"]), iFirma=getIdx(["Firma","FirmaAdi"]), iTutar=getIdx(["GenelToplam","Toplam"]), iDurum=getIdx(["Durum"]), iOlus=getIdx(["Olusturan"]);
    var from=p.from?new Date(p.from):null, to=p.to?new Date(p.to+"T23:59:59"):null;
    var rows=[];
    for(var r=1;r<vals.length;r++){
      var row=vals[r];
      if(iNo>-1&&!row[iNo]) continue;
      if(iTarih>-1&&(from||to)){ var dt=new Date(row[iTarih]); if(from&&dt&&dt<from)continue; if(to&&dt&&dt>to)continue; }
      if(p.kullaniciEmail&&iOlus>-1){if(String(row[iOlus]||"").toLowerCase()!==p.kullaniciEmail.toLowerCase())continue;}
      rows.push({ tarih:iTarih>-1?String(row[iTarih]||""):"", firma:iFirma>-1?String(row[iFirma]||""):"", tutar:iTutar>-1?Number(row[iTutar])||0:0, durum:iDurum>-1?String(row[iDurum]||""):"", olusturan:iOlus>-1?String(row[iOlus]||""):"" });
    }
    rows.sort(function(a,b){return (b.tarih||"").localeCompare(a.tarih||"");});
    return { ok:true, rows:rows };
  } catch(e) { return { ok:false, hata:e.message }; }
}

  try {
    p = p || {};

    var sh1 = tmp.getActiveSheet();
    sh1.setName("Aktivite");
    var res = api_raporKullaniciPerformans(p);

    ((res&&res.rows)||[]).forEach(function(r){ sh1.appendRow([r.tarih||"",r.firma||"",r.kullaniciEmail||"",r.kanal||"",r.islem||"",r.asama||""]); });

    var tRes = api_raporTeklifler(p);
    sh2.appendRow(["Tarih","Firma","Tutar","Durum","Oluşturan"]);
    ((tRes&&tRes.rows)||[]).forEach(function(t){ sh2.appendRow([t.tarih||"",t.firma||"",t.tutar||0,t.durum||"",t.olusturan||""]); });
    var id = tmp.getId();
    var url = "https://docs.google.com/spreadsheets/d/"+id+"/export?format=xlsx";
    var blob = UrlFetchApp.fetch(url, { headers:{Authorization:"Bearer "+ScriptApp.getOAuthToken()} }).getBlob();
    var b64 = Utilities.base64Encode(blob.getBytes());
    DriveApp.getFileById(id).setTrashed(true);
    return { ok:true, b64:b64 };
  } catch(e) { return { ok:false, hata:e.message }; }
}

  });
  var oran={olumlu:pos,olumsuz:neg,olumluOran:(pos+neg)?Math.round((pos/(pos+neg))*100):0,olumsuzOran:(pos+neg)?Math.round((neg/(pos+neg))*100):0};

// KULLANICI ÖZET
// ==========================================
function api_raporKullaniciOzet(p) {
  p = p || {};
  var ss = SpreadsheetApp.openById(CRM_SHEET_ID);
  var sh = ss.getSheetByName("HAREKETLER");
  if (!sh) return { ok:true, items:[] };

  var data = sh.getDataRange().getValues();
  if (data.length < 2) return { ok:true, items:[] };

  var head = data[0];
  function getCol(aranan) {
    for(var z=0; z<head.length; z++){
      if(String(head[z]).toLowerCase().indexOf(aranan) > -1) return z;
    }
    return -1;
  }

  var ixTar  = getCol("tarih");
  var ixUser = getCol("kullaniciemail");
  var ixKanal = getCol("kanal");
  var ixIslem = getCol("islem");

  function parseDT_(v) {
    if(v instanceof Date) return v;
    var s = String(v||"").trim();
    if(!s) return null;
    var m = s.match(/^(\d{1,2})\.(\d{1,2})\.(\d{4})/);
    if(m) return new Date(m[3], m[2]-1, m[1]);
    var d2 = new Date(s);
    return isNaN(d2.getTime()) ? null : d2;
  }

  var dFrom = p.startDate ? parseDT_(p.startDate) : new Date(2000,0,1);
  var dTo   = p.endDate   ? parseDT_(p.endDate)   : new Date(2099,11,31);
  if(dTo) dTo.setHours(23,59,59,999);

  // Dinamik kullanıcı listesi
  var map = {};

  for(var r=1; r<data.length; r++) {
    var rawTarih = data[r][ixTar];
    if(!rawTarih) continue;
    var dt = parseDT_(rawTarih);
    if(!dt || dt < dFrom || dt > dTo) continue;

    var email = String(data[r][ixUser]||"").trim().toLowerCase();
    if(!email) continue;

    if(!map[email]) {
      map[email] = {
        email: email,
        ad: email.split("@")[0].toUpperCase(),
        telefon: 0, mail: 0, toplanti: 0, teklif: 0
      };
    }

    var oKanal = String(data[r][ixKanal]||"").trim().toUpperCase();
    var oIslem = String(data[r][ixIslem]||"").trim().toUpperCase();

    if(oKanal.indexOf("TELEFON") > -1) map[email].telefon++;
    if(oKanal.indexOf("MAIL") > -1 || oKanal.indexOf("MAİL") > -1) map[email].mail++;
    if(oKanal.indexOf("TOPLANTI") > -1 || oKanal.indexOf("GÖRÜŞME") > -1) map[email].toplanti++;
    if(oIslem.indexOf("TEKLIF") > -1 || oIslem.indexOf("TEKLİF") > -1) map[email].teklif++;
  }

  var items = Object.keys(map).map(function(k){ return map[k]; });
  items.sort(function(a,b){ return (b.telefon+b.mail+b.toplanti+b.teklif) - (a.telefon+a.mail+a.toplanti+a.teklif); });

  return { ok:true, items:items };
}

function api_getPerformanceData(p) {
  try {
    p = p || {};
    var ss = SpreadsheetApp.openById(CRM_SHEET_ID);
    var sheet = ss.getSheetByName("HAREKETLER");
    if (!sheet) return [];
    var data = sheet.getDataRange().getValues();
    if (data.length < 2) return [];
    var baslik = data[0];
    function getCol(aranan){ for(var m=0;m<baslik.length;m++){if(String(baslik[m]).toLowerCase().indexOf(aranan)>-1)return m;} return -1; }
    var ixt=getCol("tarih"),ixfr=getCol("firma"),ixkl=getCol("kullaniciemail"),ixk=getCol("kanal"),ixis=getCol("islem"),ixasm=getCol("asama"),ixnot=getCol("not");
    var simdi = new Date();
    function _parsePerf(s) {
      if (!s) return null;
      s = String(s);
      var m = s.match(/(\d{1,2})\.(\d{1,2})\.(\d{4})/);
      if (m) return new Date(m[3], m[2]-1, m[1]);
      return new Date(s);
    }
    var fromDate = p.from ? _parsePerf(p.from).getTime() : new Date(simdi.getFullYear(), simdi.getMonth(), 1).getTime();
    var toDate = p.to ? (_parsePerf(p.to).getTime() + 86399000) : simdi.getTime();
    var outputArray=[], ustLimit=0;
    for(var k=data.length-1;k>=1;k--){
      var r=data[k];
      var rawDate=r[ixt];
      if(!rawDate) continue;
      var currentTDate;
      if(rawDate instanceof Date){currentTDate=rawDate;}
      else{var fixPar=String(rawDate).match(/(\d{1,2})\.(\d{1,2})\.(\d{4})/);currentTDate=fixPar?new Date(fixPar[3],fixPar[2]-1,fixPar[1]):new Date(rawDate);}
      var zamanPulu=currentTDate.getTime();
      if(zamanPulu<fromDate||zamanPulu>toDate) continue;
      var safeStringDate=Utilities.formatDate(currentTDate,"Europe/Istanbul","dd.MM.yyyy HH:mm");
      outputArray.push({tarih:safeStringDate,firma:String(r[ixfr]||"Bilinmeyen"),kullanici:String(r[ixkl]||""),kanal:String(r[ixk]||""),islem:String(r[ixis]||""),asama:String(r[ixasm]||""),not:String(r[ixnot]||"")});
      ustLimit++;
      if(ustLimit>=1500) break;
    }
    return outputArray;
  } catch(e) { return []; }
}

function ceo_parseDate_(v) {
  if(v instanceof Date) return v;
  var s=String(v||"").trim();
  if(!s) return null;
  var m=s.match(/^(\d{1,2})\.(\d{1,2})\.(\d{4})/);
  if(m) return new Date(+m[3],+m[2]-1,+m[1]);
  var d=new Date(s); return isNaN(d.getTime())?null:d;
}

function ceo_fmtDate_(v) {
  var d=ceo_parseDate_(v);
  if(!d) return String(v||"");
  return Utilities.formatDate(d,"Europe/Istanbul","dd.MM.yyyy");
}

function api_teklifDonusumOrani(p) {
  try {
    p = p || {};
    var ss = SpreadsheetApp.openById(CRM_SHEET_ID);
    var sh = ss.getSheetByName("TEKLIFLER") || ss.getSheetByName("TEKLİFLER");
    if (!sh || sh.getLastRow() < 2) return { ok:true, genel:{}, aylik:[], kisisel:[] };
    var vals = sh.getDataRange().getDisplayValues();
    var head = vals[0];
    function normKey(s) { return String(s||"").trim().toUpperCase().replace(/İ/g,"I").replace(/ı/g,"I").replace(/\s+/g,""); }
    var idx = {};
    for (var c=0; c<head.length; c++) { var k = normKey(head[c]); if (k) idx[k] = c; }
    function col(names) { for (var i=0;i<names.length;i++) { if (idx[names[i]] !== undefined) return idx[names[i]]; } return -1; }
    var iDurum = col(["DURUM"]);
    var iTarih = col(["TARIH","OLUSTURMATARIHI"]);
    var iOlus  = col(["OLUSTURAN","SENDEREMAIL"]);
    var iTutar = col(["GENELTOPLAM","TOPLAM"]);
    
    var genel = { toplam:0, onaylanan:0, reddedilen:0, bekleyen:0, toplamTutar:0, onaylananTutar:0 };
    var ayMap = {};
    var kisiMap = {};
    
    for (var r=1; r<vals.length; r++) {
      var durum = String(vals[r][iDurum] || "").toUpperCase().trim();
      if (!durum && !vals[r][iTarih]) continue;
      
      var tarihStr = String(vals[r][iTarih] || "");
      var ay = "";
      var m = tarihStr.match(/(\d{1,2})\.(\d{1,2})\.(\d{4})/);
      if (m) ay = m[3] + "-" + m[2].padStart(2,"0");
      else {
        var m2 = tarihStr.match(/(\d{4})-(\d{2})/);
        if (m2) ay = m2[1] + "-" + m2[2];
      }
      
      var olusturan = String(vals[r][iOlus] || "").toLowerCase().trim();
      var tutar = Number(String(vals[r][iTutar] || "0").replace(/\./g,"").replace(",",".")) || 0;
      
      var isOnayli = (durum === "ONAYLANDI" || durum.indexOf("KAZAN") > -1);
      var isRed = (durum.indexOf("KAYB") > -1 || durum === "REDDEDILDI" || durum === "REDDEDİLDİ");
      
      genel.toplam++;
      genel.toplamTutar += tutar;
      if (isOnayli) { genel.onaylanan++; genel.onaylananTutar += tutar; }
      else if (isRed) { genel.reddedilen++; }
      else { genel.bekleyen++; }
      
      // Aylık
      if (ay) {
        if (!ayMap[ay]) ayMap[ay] = { ay:ay, toplam:0, onaylanan:0, reddedilen:0, tutar:0, onaylananTutar:0 };
        ayMap[ay].toplam++;
        ayMap[ay].tutar += tutar;
        if (isOnayli) { ayMap[ay].onaylanan++; ayMap[ay].onaylananTutar += tutar; }
        else if (isRed) { ayMap[ay].reddedilen++; }
      }
      
      // Kişisel
      if (olusturan) {
        if (!kisiMap[olusturan]) kisiMap[olusturan] = { email:olusturan, ad:olusturan.split("@")[0].toUpperCase(), toplam:0, onaylanan:0, reddedilen:0, tutar:0, onaylananTutar:0 };
        kisiMap[olusturan].toplam++;
        kisiMap[olusturan].tutar += tutar;
        if (isOnayli) { kisiMap[olusturan].onaylanan++; kisiMap[olusturan].onaylananTutar += tutar; }
        else if (isRed) { kisiMap[olusturan].reddedilen++; }
      }
    }
    
    genel.donusumOrani = genel.toplam > 0 ? Math.round((genel.onaylanan / genel.toplam) * 100) : 0;
    
    var aylik = Object.keys(ayMap).map(function(k) { 
      var o = ayMap[k]; 
      o.donusumOrani = o.toplam > 0 ? Math.round((o.onaylanan / o.toplam) * 100) : 0; 
      return o; 
    }).sort(function(a,b) { return b.ay.localeCompare(a.ay); });
    
    var kisisel = Object.keys(kisiMap).map(function(k) { 
      var o = kisiMap[k]; 
      o.donusumOrani = o.toplam > 0 ? Math.round((o.onaylanan / o.toplam) * 100) : 0; 
      return o; 
    }).sort(function(a,b) { return b.onaylanan - a.onaylanan; });
    
    return { ok:true, genel:genel, aylik:aylik.slice(0,12), kisisel:kisisel };
  } catch(e) { return { ok:false, msg:e.message }; }
}

function api_gunlukKpiGetir(p) {

    function getCol(aranan) {
      for (var z=0; z<head.length; z++) {
        if (String(head[z]).toLowerCase().replace(/\s+/g,"").indexOf(aranan) > -1) return z;
      }
      return -1;
    }
    var ixt = getCol("tarih"), ixUser = getCol("kullaniciemail"), ixKanal = getCol("kanal"), ixIslem = getCol("islem");
    
    // Bugünün tarihi
    var bugun = Utilities.formatDate(new Date(), "Europe/Istanbul", "dd.MM.yyyy");
    var hedefTarih = String(p && p.tarih || bugun);
    
    // ROL haritası
    var roller = {};
    try {
      var rolHaritasi = ROL_HARITASI || {};
      for (var email in rolHaritasi) {
        roller[email.toLowerCase()] = rolHaritasi[email];
      }
    } catch(e) {}

    var shPer = ss.getSheetByName("PERSONEL_IK");
    var personelMap = {};
    if (shPer && shPer.getLastRow() > 1) {
      var perData = shPer.getDataRange().getValues();
      for (var pi=1; pi<perData.length; pi++) {
        var pEmail = String(perData[pi][2] || "").toLowerCase().trim();
        var pAd = String(perData[pi][1] || "").trim();
        if (pEmail) personelMap[pEmail] = pAd;
      }
    }
    
    // Bugünkü aktiviteleri say
    var kisiMap = {};
    for (var i=1; i<data.length; i++) {
      var rawTarih = data[i][ixt];
      var tarihStr = "";
      if (rawTarih instanceof Date) {
        tarihStr = Utilities.formatDate(rawTarih, "Europe/Istanbul", "dd.MM.yyyy");
      } else {
        tarihStr = String(rawTarih || "").substring(0, 10);
      }
      if (tarihStr !== hedefTarih) continue;
      
      var email = String(data[i][ixUser] || "").toLowerCase().trim();
      if (!email) continue;
      var kanal = String(data[i][ixKanal] || "").toUpperCase().trim();
      var islem = String(data[i][ixIslem] || "").toUpperCase().trim();
      
      if (kanal === "SİSTEM" || kanal === "SISTEM") continue;
      
      if (!kisiMap[email]) {
        kisiMap[email] = { email:email, ad:"", rol:"", telefon:0, mail:0, toplanti:0, teklif:0, tanitim:0, diger:0, toplam:0 };
      }
      kisiMap[email].toplam++;
      
      if (kanal.indexOf("TELEFON") > -1 || kanal.indexOf("ARAMA") > -1) kisiMap[email].telefon++;
      else if (kanal.indexOf("MAIL") > -1 || kanal.indexOf("MAİL") > -1) kisiMap[email].mail++;
      else if (kanal.indexOf("TOPLANTI") > -1 || kanal.indexOf("GÖRÜŞME") > -1 || kanal.indexOf("GORUSME") > -1) kisiMap[email].toplanti++;
      else kisiMap[email].diger++;
      
      if (islem.indexOf("TEKLIF") > -1 || islem.indexOf("TEKLİF") > -1) kisiMap[email].teklif++;
      if (islem.indexOf("TANITIM") > -1) kisiMap[email].tanitim++;
    }
    
    // Tüm PAZARLAMA personelini ekle (bugün aktivite olmayanlar dahil)
    for (var em in roller) {
      if (roller[em] === "PAZARLAMA" && !kisiMap[em]) {
        kisiMap[em] = { email:em, ad:"", rol:"PAZARLAMA", telefon:0, mail:0, toplanti:0, teklif:0, tanitim:0, diger:0, toplam:0 };
      }
    }
    
    // Ad ve rol eşleştir, hedef hesapla
    var kisiler = [];
    for (var k in kisiMap) {
      var o = kisiMap[k];
      o.ad = personelMap[k] || k.split("@")[0].toUpperCase();
      o.rol = roller[k] || "PERSONEL";
      Logger.log("KPI lookup: " + k + " -> " + JSON.stringify(KISI_HEDEFLERI[k]));
      var hedef = GUNLUK_HEDEFLER[o.rol] || GUNLUK_HEDEFLER["PERSONEL"];
      o.hedefler = hedef;
      
      // Yüzde hesapla
      var toplamHedef = (hedef.telefon + hedef.mail + hedef.toplanti + hedef.teklif + hedef.tanitim) || 1;
      var toplamYapilan = o.telefon + o.mail + o.toplanti + o.teklif + o.tanitim;
      o.yuzdeTamamlama = Math.min(100, Math.round((toplamYapilan / toplamHedef) * 100));

      o.eksikler = [];
      if (hedef.telefon > 0 && o.telefon < hedef.telefon) o.eksikler.push("Telefon: " + o.telefon + "/" + hedef.telefon);
      if (hedef.mail > 0 && o.mail < hedef.mail) o.eksikler.push("Mail: " + o.mail + "/" + hedef.mail);

      if (hedef.teklif > 0 && o.teklif < hedef.teklif) o.eksikler.push("Teklif: " + o.teklif + "/" + hedef.teklif);
      if (hedef.tanitim > 0 && o.tanitim < hedef.tanitim) o.eksikler.push("Tanıtım: " + o.tanitim + "/" + hedef.tanitim);
      
      o.hedefeUlasti = o.eksikler.length === 0;
      kisiler.push(o);
    }
    
    kisiler.sort(function(a,b) { return a.yuzdeTamamlama - b.yuzdeTamamlama; });
    
    return { ok:true, tarih:hedefTarih, kisiler:kisiler };
  } catch(e) { return { ok:false, msg:e.message }; }
}

  var head = vals[0];
  var idx = {};
  head.forEach(function(h, i) { idx[String(h || "").toLowerCase().replace(/\s+/g, "").trim()] = i; });

  var ixTarih  = idx["tarih"];
  var ixFirma  = idx["firmaunvani"];
  var ixFirmaId= idx["firmaid"];
  var ixUser   = idx["kullaniciemail"];
  var ixKanal  = idx["kanal"];
  var ixIslem  = idx["islem"];
  var ixAsama  = idx["asama"];
  var ixNot    = idx["not"];

  function s(x) { return String(x == null ? "" : x).trim(); }

  function parseTR_(t) {
    t = s(t).split(" ")[0];
    var p = t.split(".");
    if (p.length < 3) return null;
    return new Date(+p[2], +p[1]-1, +p[0]);
  }

  // Tarih filtresi
  var now = new Date();
  var d1, d2;
  var aralik = s(opts.aralik || "bugun");

  if (aralik === "bugun") {
    d1 = new Date(now.getFullYear(), now.getMonth(), now.getDate());
    d2 = new Date(now.getFullYear(), now.getMonth(), now.getDate(), 23, 59, 59);
  } else if (aralik === "buhafta") {
    var gun = now.getDay() || 7;
    d1 = new Date(now); d1.setDate(now.getDate() - gun + 1); d1.setHours(0,0,0,0);
    d2 = new Date(now); d2.setHours(23,59,59,999);
  } else if (aralik === "buay") {
    d1 = new Date(now.getFullYear(), now.getMonth(), 1);
    d2 = new Date(now.getFullYear(), now.getMonth(), now.getDate(), 23, 59, 59);
  } else if (opts.from && opts.to) {
    d1 = parseTR_(opts.from); d2 = parseTR_(opts.to);
    if (d2) d2.setHours(23,59,59,999);
  }

  var fUser = s(opts.kullaniciEmail).toLowerCase();

  // Personel bazında sayaçlar
  var personeller = {};

  function getP(email) {
    if (!personeller[email]) {
      personeller[email] = {
        email: email,
        firmaAcma: 0,
        tanıtimMail: 0,
        telefon: 0,
        whatsapp: 0,
        teklif: 0,
        teklifTutar: 0,
        onay: 0,
        red: 0,
        firmalar: {}
      };
    }
    return personeller[email];
  }

  // Detay listesi
  var detaylar = [];

  for (var i = 1; i < vals.length; i++) {
    var r = vals[i];
    var tarih = parseTR_(r[ixTarih]);
    if (d1 && tarih && tarih < d1) continue;
    if (d2 && tarih && tarih > d2) continue;

    var user   = s(r[ixUser]).toLowerCase();
    var kanal  = s(r[ixKanal]).toUpperCase();
    var islem  = s(r[ixIslem]).toUpperCase();
    var asama  = s(r[ixAsama]).toUpperCase();
    var firma  = s(r[ixFirma]);
    var firmaId= s(r[ixFirmaId]);
    var not    = s(r[ixNot]);
    var tarihStr = s(r[ixTarih]);

    if (fUser && user !== fUser) continue;
    if (!user) continue;

    var p = getP(user);

    // Firma açma
    var islemNorm = islem.replace(/\s+/g,"_").replace(/İ/g,"I").replace(/Ğ/g,"G").replace(/Ü/g,"U").replace(/Ş/g,"S").replace(/Ö/g,"O").replace(/Ç/g,"C");
    if (islemNorm === "FIRMA_KAYIT" || islemNorm === "YENI_KAYIT" || islemNorm === "FIRMA_EKLE") {
      p.firmaAcma++;
      p.firmalar[firmaId || firma] = firma;
    }

    // Tanıtım maili
    if (kanal === "MAIL" && (islemNorm === "TANITIM" || islemNorm === "ILETISIM")) {
      p.tanıtimMail++;
    }

    // Telefon
    if (kanal === "TELEFON") {
      p.telefon++;
    }

    // WhatsApp
    if (kanal === "WHATSAPP") {
      p.whatsapp++;
    }

    // Teklif
    if (islemNorm === "TEKLIF") {
      p.teklif++;
    }

    // Onay / Red
    if (islemNorm === "TEKLIF_ONAY" || asama === "ONAYLANDI" || asama === "KAZANILDI") {
      p.onay++;
    }
    if (asama === "REDDEDILDI" || asama === "KAYBEDILDI") {
      p.red++;
    }

    detaylar.push({
      tarih: tarihStr,
      user: user,
      firma: firma,
      kanal: kanal,
      islem: islem,
      asama: asama,
      not: not
    });
  }

  // Sonuç listesi
  var liste = [];
  for (var email in personeller) {
    var p = personeller[email];
    var donusum = p.teklif > 0 ? Math.round((p.onay / p.teklif) * 100) : 0;
    liste.push({
      email: p.email,
      ad: p.email.split("@")[0].replace(/\./g," ").toUpperCase(),
      firmaAcma: p.firmaAcma,
      tanıtimMail: p.tanıtimMail,
      telefon: p.telefon,
      whatsapp: p.whatsapp,
      teklif: p.teklif,
      onay: p.onay,
      red: p.red,
      donusum: donusum
    });
  }

  // Toplam satırı
  var toplam = { email:"TOPLAM", ad:"TOPLAM", firmaAcma:0, tanıtimMail:0, telefon:0, whatsapp:0, teklif:0, onay:0, red:0, donusum:0 };
  liste.forEach(function(x) {
    toplam.firmaAcma  += x.firmaAcma;
    toplam.tanıtimMail+= x.tanıtimMail;
    toplam.telefon    += x.telefon;
    toplam.whatsapp   += x.whatsapp;
    toplam.teklif     += x.teklif;
    toplam.onay       += x.onay;
    toplam.red        += x.red;
  });
  toplam.donusum = toplam.teklif > 0 ? Math.round((toplam.onay / toplam.teklif) * 100) : 0;

  return { ok: true, liste: liste, toplam: toplam, detaylar: detaylar, aralik: aralik };
}
