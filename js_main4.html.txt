var _rotaTumFirmalar = [];   // tüm veri burada tutulur
var _rotaGosterilenSayisi = 15;  // başlangıçta 15 göster

function rotaListesiniGetir() {
  var box = document.getElementById("rotaFirmaListesi");
  if (!box) return;
  box.innerHTML = "<div style='color:#f59e0b;text-align:center;font-weight:bold;grid-column:1/-1;padding:20px;'>⏳ Firmalar yükleniyor...<\/div>";

  google.script.run
    .withSuccessHandler(function(data) {
      _rotaTumFirmalar = (data || []).filter(function(f) {
        return f.FirmaAdi || f.firma;
      });
      _rotaGosterilenSayisi = 15;
      document.getElementById("rotaArama").value = "";
      rotaFiltrele();
    })
    .withFailureHandler(function(err) {
      box.innerHTML = "<div style='color:red;text-align:center;grid-column:1/-1;padding:20px;'>Hata: " + err.message + "<\/div>";
    })
    .api_firmalarListe();
}

function rotaFiltrele() {
  var ara = (document.getElementById("rotaArama").value || "").toLowerCase().trim();
  var filtreli = _rotaTumFirmalar.filter(function(f) {
    var ad = String(f.FirmaAdi || f.firma || "").toLowerCase();
    var ilce = String(f.Ilce || f.ilce || "").toLowerCase();
    return !ara || ad.indexOf(ara) > -1 || ilce.indexOf(ara) > -1;
  });
  _rotaGosterilenSayisi = 15;
  rotaRender(filtreli);
}

function api_personelSil(personelId) {
  try {
    var ss = SpreadsheetApp.openById(CRM_SHEET_ID);
    var sh = ss.getSheetByName("PERSONEL_IK");
    if (!sh) return { ok: false, msg: "PERSONEL_IK bulunamadı" };
    var data = sh.getDataRange().getValues();
    for (var i = 1; i < data.length; i++) {
      if (String(data[i][0]) === String(personelId)) {
        sh.deleteRow(i + 1);
        // Kullanicilar'dan da sil
        var ku = ss.getSheetByName("Kullanicilar");
        if (ku) {
          var kd = ku.getDataRange().getValues();
          for (var j = 1; j < kd.length; j++) {
            if (String(kd[j][0]).toLowerCase() === String(data[i][2]).toLowerCase()) {
              ku.deleteRow(j + 1);
              break;
            }
          }
        }
        chatBildir_("🗑️ *Personel Silindi*\nAd: " + String(data[i][1]||""));
        return { ok: true };
      }
    }
    return { ok: false, msg: "Personel bulunamadı" };
  } catch(e) { return { ok: false, msg: e.message }; }
}

function rotaRender(liste) {
  var box = document.getElementById("rotaFirmaListesi");
  var sayacEl = document.getElementById("rotaSayac");
  var dahaFazlaDiv = document.getElementById("rotaDahaFazlaDiv");
  if (!box) return;

  if (!liste.length) {
    box.innerHTML = "<div style='color:#ef4444;text-align:center;grid-column:1/-1;padding:30px;'>Firma bulunamadı.<\/div>";
    if (sayacEl) sayacEl.textContent = "0 firma";
    if (dahaFazlaDiv) dahaFazlaDiv.style.display = "none";
    return;
  }

  // Gösterilecek kadar kes
  var gosterilen = liste.slice(0, _rotaGosterilenSayisi);
  if (sayacEl) sayacEl.textContent = liste.length + " firma" + (liste.length > _rotaGosterilenSayisi ? " (" + _rotaGosterilenSayisi + " gösteriliyor)" : "");

  box.innerHTML = gosterilen.map(function(f) {
    var firmaAdi = f.FirmaAdi || f.firma || "İsimsiz";
    var adres    = f.Adres || f.adres || "";
    var ilce     = f.Ilce || f.ilce || "";
    var sorumlu  = (f.SorumluEmail || "").split("@")[0].toUpperCase() || "-";
    var searchStr = encodeURIComponent(firmaAdi + (adres ? " " + adres : "") + (ilce ? " " + ilce : ""));

    return '<label style="display:flex;gap:12px;align-items:flex-start;padding:14px;border:1px solid #cbd5e1;border-radius:10px;cursor:pointer;background:#fff;" onmouseover="this.style.borderColor=\'#0f766e\'" onmouseout="this.style.borderColor=\'#cbd5e1\'">' +
      '<input type="checkbox" class="rotaFirmaChk" value="' + searchStr + '" style="transform:scale(1.5);cursor:pointer;margin-top:3px;">' +
      '<div>' +
        '<div style="font-weight:900;color:#1e293b;font-size:14px;">' + window.esc(firmaAdi) + '<\/div>' +
        (ilce ? '<div style="font-size:11px;color:#0f766e;margin-top:2px;">📍 ' + window.esc(ilce) + '<\/div>' : '') +
        '<div style="font-size:11px;color:#64748b;margin-top:2px;">👤 ' + window.esc(sorumlu) + '<\/div>' +
      '<\/div>' +
    '<\/label>';
  }).join("");

  // Daha fazla butonu
  if (dahaFazlaDiv) {
    dahaFazlaDiv.style.display = liste.length > _rotaGosterilenSayisi ? "block" : "none";
    dahaFazlaDiv._rotaListe = liste; // listeyi sakla
  }
}

function googleHaritalardaRotaCiz() {
  var chks = document.querySelectorAll(".rotaFirmaChk:checked");
  if(chks.length === 0) { 
  alert("🚨 Rota çizmek için en az 1 firma seçmelisiniz."); 
  return; 
  }
  
  var baseUrl = "https://www.google.com/maps/dir/?api=1";
  var waypoints = [];
  var destination = "";

  chks.forEach(function(chk, i) {
  if(i === chks.length - 1) {
  destination = chk.value;
  } else {
  waypoints.push(chk.value);
  }
  });

  var finalUrl = baseUrl + "&destination=" + destination;
  if(waypoints.length > 0) {
  finalUrl += "&waypoints=" + waypoints.join("|");
  }

  window.open(finalUrl, "_blank");
}

window.handleEvrakUpload = function() {
  var fileInput = document.getElementById("evrakFileInput");

  if (!file) return;

  var fAdi = String(window._selectedFirma || "Bilinmiyor").trim();

  alert("🚨 Dosya yüklemeden önce firmanın sistemde kayıtlı (ID'li) olduğundan emin olun.");
  return;
  }

  if (file.size > 5 * 1024 * 1024) {
  alert("❌ Dosya boyutu 5 MB'dan küçük olmalıdır.");
  return;
  }

  notify("⏳ Dosya Drive'a yükleniyor, lütfen bekleyin...", "info");

  var reader = new FileReader();
  reader.onload = function(e) {
  var base64Data = e.target.result.split(',')[1]; 

  var payload = {
  firmaId: fId,
  firmaAdi: fAdi,
  fileName: file.name,
  mimeType: file.type,
  base64Data: base64Data,
  kullanici: String(window.currentUserEmail || window.aktifKullanici || "")
  };

  google.script.run
  .withSuccessHandler(function(res) {
  if (res && res.ok) {
  notify(res.msg, "success");
  evraklariGoster(); 
  } else {
  notify("Hata: " + (res ? res.msg : "Bilinmeyen"), "error");
  }
  })
  .withFailureHandler(function(err) {
  notify("Bağlantı Hatası: " + err.message, "error");
  })
  .api_evrakYukle(payload);
  };
  reader.readAsDataURL(file);
};

window.setPerfDateRange = function(tip) {
  var fromInput = document.getElementById("perf_from");
  var toInput = document.getElementById("perf_to");
  if(!fromInput || !toInput) return;

  var bugun = new Date();
  var fromDate = new Date();
  var toDate = new Date();

  if (tip === 'gunluk') {
  fromDate = bugun;
  toDate = bugun;
  } 
  else if (tip === 'haftalik') {
  var day = bugun.getDay();
  var diff = bugun.getDate() - day + (day === 0 ? -6 : 1); 
  fromDate = new Date(bugun.setDate(diff));
  toDate = new Date(); 
  } 
  else if (tip === 'aylik') {
  fromDate = new Date(bugun.getFullYear(), bugun.getMonth(), 1);
  toDate = new Date(bugun.getFullYear(), bugun.getMonth() + 1, 0); 
  }
  var formatla = function(d) {
  var ay = ("0" + (d.getMonth() + 1)).slice(-2);
  var gun = ("0" + d.getDate()).slice(-2);
  return d.getFullYear() + "-" + ay + "-" + gun;
  };

  fromInput.value = formatla(fromDate);
  toInput.value = formatla(toDate);
  if (typeof loadPerformanceDashboard === "function") {
  loadPerformanceDashboard();
  }
};

function finTipDegisti() {
  if (!document.getElementById('finTip')) return;
  var tip = document.getElementById('finTip').value;
  var firmaDiv = document.getElementById('finFirmaDiv');
  var kisiDiv  = document.getElementById('finKisiDiv');
  var gelirMi  = tip.indexOf('GELİR_ISG') > -1 || tip.indexOf('GELİR_MOBİL') > -1 || tip.indexOf('GELİR_ISE') > -1;
  if (firmaDiv) firmaDiv.style.display = gelirMi ? 'block' : 'none';
  if (kisiDiv)  kisiDiv.style.display  = gelirMi ? 'block' : 'none';

  // Autocomplete sadece gelir tiplerinde aktif
  if (gelirMi && !window._finFirmaAutoBound) {
    window._finFirmaAutoBound = true;
    var inp = document.getElementById('finFirma');
    if (!inp) return;
    // Firma listesini yükle
    if (!window._finFirmaListesi) {
      google.script.run
        .withSuccessHandler(function(list) {
          window._finFirmaListesi = list || [];
        })
        .api_firmaAdlariGetir();
    }
    inp.addEventListener('input', function() {
      var val = this.value.toLowerCase();
      var oneri = document.getElementById('finFirmaOneri');
      if (!oneri) return;
      if (!val || val.length < 2) { oneri.style.display = 'none'; return; }
      var eslesme = (window._finFirmaListesi || []).filter(function(f) {
        return f.toLowerCase().indexOf(val) > -1;
      }).slice(0, 6);
      if (!eslesme.length) { oneri.style.display = 'none'; return; }
      oneri.innerHTML = eslesme.map(function(f) {
        return "<div onclick=\"document.getElementById('finFirma').value='" + f.replace(/'/g,"\\'") + "';document.getElementById('finFirmaOneri').style.display='none';\" style='padding:8px 12px;cursor:pointer;border-bottom:1px solid #f1f5f9;font-size:13px;' onmouseover=\"this.style.background='#f0fdf4'\" onmouseout=\"this.style.background=''\">🏢 " + f + "<\/div>";
      }).join('');
      oneri.style.display = 'block';
    });
  }
}

function kasaDurumunuGuncelle() {
  var bakiyeKutusu = document.getElementById("finansKasaOzet");
  if (bakiyeKutusu) bakiyeKutusu.innerHTML = "⏳";

  google.script.run
    .withSuccessHandler(function(res) {
      if (!res || !res.ok) {
        if (bakiyeKutusu) bakiyeKutusu.innerHTML = "0,00 ₺";
        return;
      }
      var fmt = function(n) { return Number(n||0).toLocaleString('tr-TR', {minimumFractionDigits:2}) + " ₺"; };

      // Toplam bakiye
      var bakiye = res.toplam || 0;
      if (bakiyeKutusu) {
        bakiyeKutusu.innerHTML = bakiye < 0
          ? "<span style='color:#fca5a5;'>" + fmt(bakiye) + "<\/span>"
          : fmt(bakiye);
      }

      // Bu ay gelir
      var gelirEl = document.getElementById('kasaAyGelir');
      if (gelirEl) gelirEl.innerHTML = fmt(res.ayGelir || 0);

      // Bu ay gider
      var giderEl = document.getElementById('kasaAyGider');
      if (giderEl) giderEl.innerHTML = fmt(res.ayGider || 0);

      // Bu ay net
      var netEl = document.getElementById('kasaAyNet');
      if (netEl) {
        var net = (res.ayGelir || 0) - (res.ayGider || 0);
        netEl.innerHTML = "<span style='color:" + (net >= 0 ? '#15803d' : '#b91c1c') + ";'>" + (net >= 0 ? '+' : '') + fmt(net) + "<\/span>";
      }
    })
    .withFailureHandler(function(err) {
      if (bakiyeKutusu) bakiyeKutusu.innerHTML = "Hata!";
    })
    .api_getToplamKasa();
}

function filtreUyarilar(modulTipi) {

  // Çözülmüş / okunmuş olanları gizle
  filtreli = filtreli.filter(function(x) { return !x.okundu; });

  if (modulTipi !== 'TÜMÜ') {
    filtreli = filtreli.filter(function(x) { return String(x.tip||"").indexOf(modulTipi) > -1; });
  }

  if (filtreli.length === 0) {
    tbody.innerHTML = "<tr><td colspan='5' style='text-align:center;padding:30px;font-weight:bold;color:#10b981;'>✅ Harika! Bu kategoride bekleyen uyarı yok.<\/td><\/tr>";
    return;
  }

  tbody.innerHTML = filtreli.map(function(r) {
    var drmRenk = String(r.tip||"").indexOf("ALARM") > -1 ? "#fee2e2; color:#991b1b;" :
      (String(r.tip||"").indexOf("MÜJDE") > -1 ? "#dcfce7; color:#166534;" : "#f1f5f9; color:#475569;");

    var mdlRenk = String(r.tip||"").indexOf("FİNANS") > -1 ? "#fef08a; color:#854d0e;" :

    // PERSONEL talebi → Onayla / Reddet butonu
    var islemSutunu;

      islemSutunu =
        '<div style="display:flex;flex-direction:column;gap:6px;align-items:center;width:100%;min-width:120px;">' +

function aktifPersonelListesiniCek() {
  var tbody = document.getElementById("personelListesiBody");
  if(!tbody) return;
  
  tbody.innerHTML = "<tr><td colspan='7' style='text-align:center; padding:20px; font-weight:bold; color:#2563eb;'>Yükleniyor... ⏳<\/td><\/tr>";
  
  google.script.run
  .withSuccessHandler(function(response) {
    // 1. GELEN VERİYİ KONTROL ET VE DİZİYE ZORLA
    var asilListe = [];
    if (Array.isArray(response)) {
      asilListe = response;
    } else if (response && response.liste && Array.isArray(response.liste)) {
      asilListe = response.liste;
    }

    // Cache'e diziyi at
    window._personelCache = asilListe;

    // 2. HATA MESAJI VAR MI?
    if (response && response.hata) {
      tbody.innerHTML = "<tr><td colspan='7' style='text-align:center; padding:20px; color:#ef4444;'>❌ " + response.msg + "<\/td><\/tr>";
      return;
    }

    // 3. LİSTE BOŞ MU?
    if (asilListe.length === 0) {
      tbody.innerHTML = "<tr><td colspan='7' style='text-align:center; padding:20px; color:#64748b;'>Kayıtlı personel bulunamadı veya yetkiniz yok.<\/td><\/tr>";
      return;
    }

    // 4. TABLOYU ÇİZ (Artık 'asilListe' kesinlikle bir dizidir, hata vermez)
    personelTabloRender(asilListe);
  })
  .withFailureHandler(function(err) {
    tbody.innerHTML = "<tr><td colspan='7' style='text-align:center; color:red; padding:20px;'>Sunucu Hatası: " + err.message + "<\/td><\/tr>";
  })
  .api_personelListesiGetir(window.currentUserEmail);
}

function parseTRDate_(s) {
  var str = String(s || "").trim();
  var m = str.match(/^(\d{1,2})\.(\d{1,2})\.(\d{4})/);
  if (m) return new Date(+m[3], +m[2]-1, +m[1]).getTime();
  var d = new Date(str);
  return isNaN(d) ? 0 : d.getTime();
}

function isgHizmetBilgisiDoldur(firmaId) {
  if (!firmaId) return;
  google.script.run
  .withSuccessHandler(function(res) {
  if (!res) return;
  var el = document.getElementById("e_tehlikeSinifi");
  if (el) el.value = res.tehlikeSinifi || "";
  var el2 = document.getElementById("e_iguSinifi");
  if (el2) el2.value = res.iguSinifi || "C";
  var el3 = document.getElementById("e_sozlesmeBas");
  if (el3) el3.value = res.sozlesmeBas || "";
  var el4 = document.getElementById("e_sozlesmeBit");
  if (el4) el4.value = res.sozlesmeBit || "";

  var hizmetler = String(res.hizmetler || "");
  ["osgb","IGU","Hekim","DSP","Mobil","IseGiris"].forEach(function(h) {
  var cb = document.getElementById("e_hiz_" + h);
  if (cb) cb.checked = hizmetler.indexOf(cb.value) > -1;
  });
  isgSureOzetiGoster(res);
  })
  .api_isgHizmetGetir(firmaId);
}

function isgSureOzetiGoster(res) {
  var ozet = document.getElementById("isgSureOzeti");
  if (!ozet || !res) return;
  var satirlar = [];
  if (res.aylikIGU  > 0) satirlar.push("🦺 İGU: <b>" + res.aylikIGU + " dk/ay<\/b>");
  if (res.aylikHekim > 0) satirlar.push("🩺 Hekim: <b>" + res.aylikHekim + " dk/ay<\/b>");
  if (res.aylikDSP  > 0) satirlar.push("💊 DSP: <b>" + res.aylikDSP + " dk/ay<\/b>");
  if (satirlar.length > 0) {
  ozet.innerHTML = "📊 Aylık Zorunlu Süre → " + satirlar.join(" &nbsp;|&nbsp; ");
  ozet.style.display = "block";
  } else {
  ozet.style.display = "none";
  }
}

function isgHizmetKaydetUI() {
  var firmaId  = window._selectedFirmaId || "";
  var firmaAdi = (document.getElementById("e_firma") || {}).value || "";
  var tehlike  = (document.getElementById("e_tehlikeSinifi") || {}).value || "";
  var iguSinifi = (document.getElementById("e_iguSinifi") || {}).value || "C";
  var sozBas  = (document.getElementById("e_sozlesmeBas") || {}).value || "";
  var sozBit  = (document.getElementById("e_sozlesmeBit") || {}).value || "";
  var calisan  = parseInt((document.getElementById("e_calisanSayisi") || {}).value) || 0;

  if (!firmaId || !tehlike) {
  alert("Firma ID ve tehlike sınıfı zorunlu!");
  return;
  }

  var hizmetler = [];
  ["IGU","Hekim","DSP","Mobil","IseGiris"].forEach(function(h) {
  var cb = document.getElementById("e_hiz_" + h);
  if (cb && cb.checked) hizmetler.push(cb.value);
  });

  if (hizmetler.length === 0) {
  alert("En az bir hizmet seçin!");
  return;
  }

  var mesaj = document.getElementById("isgKayitMesaj");
  if (mesaj) mesaj.textContent = "Kaydediliyor...";

  google.script.run
  .withSuccessHandler(function(res) {
  if (res && res.ok) {
  if (mesaj) mesaj.textContent = "✅ Kaydedildi!";
  isgSureOzetiGoster(res);
  setTimeout(function() { if (mesaj) mesaj.textContent = ""; }, 3000);
  } else {
  if (mesaj) mesaj.textContent = "❌ " + (res ? res.msg : "Hata");
  }
  })
  .withFailureHandler(function(err) {
  if (mesaj) mesaj.textContent = "❌ " + err.message;
  })
  .api_isgHizmetKaydet({
  firmaId:  firmaId,
  firmaAdi:  firmaAdi,
  tehlikeSinifi: tehlike,
  calisanSayisi: calisan,
  iguSinifi:  iguSinifi,
  hizmetler:  hizmetler,
  sozlesmeBas:  sozBas,
  sozlesmeBit:  sozBit,
  durum:  "AKTİF"
  });
}
