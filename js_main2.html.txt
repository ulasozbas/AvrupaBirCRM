function indir(){
  if (!validateTeklif()) return;

  var p = payload();

  var elFirma = document.getElementById("firma");
  var firmaRaw = String(elFirma && elFirma.value ? elFirma.value : "").trim();

  var firmaSafe = firmaRaw
  .replace(/[\\\/:*?"<>|]/g, "-")
  .replace(/\s+/g, " ")
  .trim();

  p.fileName = (firmaSafe ? firmaSafe : "teklif") + ".pdf";

log("PDF hazırlanıyor.");

  p.teklifId = p.teklifId || (p.meta && p.meta.teklifId) || "";

  google.script.run
  .withSuccessHandler(function(r){
  if (!r || !r.b64){
  log("Hata ❌ PDF verisi gelmedi");
  return;
  }

  var byteChars = atob(r.b64);
  var byteNumbers = new Array(byteChars.length);
  for (var i=0; i<byteChars.length; i++) byteNumbers[i] = byteChars.charCodeAt(i);
  var blob = new Blob([new Uint8Array(byteNumbers)], {type:'application/pdf'});

  var a = document.createElement('a');
  a.href = URL.createObjectURL(blob);
  a.download = (r && r.name) ? r.name : p.fileName;

  document.body.appendChild(a);
  a.click();
  a.remove();

  log("OK ✅ İndirme başladı.");
  })
  .withFailureHandler(function(e){
  log((e && e.message) ? e.message : String(e));
  })
  .api_pdfBase64(p); 
}

window.esc = function(s){
return String(s ?? "").replace(/[&<>"']/g, m => ({
  "&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"
}[m]));
}

window._firmalarMaster = [];

function buildFirmaCacheFromTeklifGecmisi() {
  var master = window._firmalarMaster || [];
  var map = new Map();

  master.forEach(function(f) {
  var firmaAdi = String(f.FirmaAdi || f.FirmaUnvani || "").trim();
  if (!firmaAdi) return;
  map.set(firmaAdi, {
  firma: firmaAdi,
  firmaId: String(f.FirmaID || "").trim(),
  yetkili: String(f.YetkiliAdi || f.Yetkili || "").trim(),
  iletisim: [String(f.Telefon || "").trim(), String(f.Email || "").trim()].filter(Boolean).join(" • "),
  sonIslem: "", kaydeden: "", source: "MASTER", _lastDate: 0
  });
  });

  var src = (typeof _teklifGecmisCache !== "undefined" ? _teklifGecmisCache : window._teklifGecmisCache) || [];
  src.forEach(function(r) {
  var fAdi = String(r.FirmaAdi || r.firma || "").trim();
  if (!fAdi) return;
  if (!map.has(fAdi)) {
  map.set(fAdi, { firma: fAdi, firmaId: String(r.FirmaID || "").trim(), source: "TEKLIF_GECMISI", _lastDate: 0 });
  }
  var curr = map.get(fAdi);
  var d = (typeof parseTRDate_ === "function") ? parseTRDate_(r.tarih) : 0;
  if (d > curr._lastDate) {
  curr._lastDate = d;
  curr.sonIslem = r.teklifNo ? ("Teklif " + r.teklifNo) : "Teklif";
  curr.kaydeden = r.olusturan || "";
  }
  });

  window._firmaCache = Array.from(map.values()).sort(function(a, b) { return String(a.firma).localeCompare(String(b.firma), "tr"); });
}

function norm(s) { 
  return String(s || "").toLocaleLowerCase("tr-TR").trim(); 
}

function mergeFirmalarWithGecmis() {
  var a = Array.isArray(window._firmaCache) ? window._firmaCache : [];
  var g = Array.isArray(window._teklifGecmisCache) ? window._teklifGecmisCache : [];

  var seen = new Set(a.map(function(x) { return norm(x.firma); }));

  var extra = [];
  for (var i = 0; i < g.length; i++) {
  var r = g[i];
  var firmaAdi = String(r.firma || r.FirmaAdi || r.FirmaUnvani || r.firmaAdi || "").trim();
  if (!firmaAdi) continue;
  var key = norm(firmaAdi);
  if (seen.has(key)) continue;
  seen.add(key);
  var yetkiliAdi = String(r.yetkili || r.YetkiliAdi || r.yetkiliAdi || "").trim();
  extra.push({
  firma: firmaAdi,
  firmaId: "",
  yetkili: yetkiliAdi,
  iletisim: "-",
  sonIslem: String(r.tarih || r.Tarih || "").trim() || "-",
  kaydeden: String(r.olusturan || r.Olusturan || "").trim() || "-",
  source: "ESKI_TEKLIF",
  _lastDate: new Date(0)
  });
  }

  window._firmaCache = a.concat(extra);
  log("mergeFirmalarWithGecmis OK - extra=" + extra.length + " total=" + window._firmaCache.length);
}

function loadTeklifGecmisi(done){

function policeAramaFiltrele() {

function applyGecmisFilterSortRender(){
  var q = String(document.getElementById("gecFiltre")?.value || "").trim().toLowerCase();
  var sort = String(document.getElementById("gecSort")?.value || "tarih_desc");
  var list = (window._teklifGecmisCache || []).slice();

  if (q){
  list = list.filter(function(r){
  var s = [r.teklifNo, r.firma, r.yetkili, r.mailTo, r.durum].join(" ").toLowerCase();
  return s.indexOf(q) >= 0;
  });
  }

  function parseTRMoney(x){
  var s = String(x||"").replace(/\s/g,"");
  if (!s) return 0;
  s = s.replace(/\./g,"").replace(",",".");
  var n = Number(s);
  return isFinite(n) ? n : 0;
  }

  function parseTRDate(x){
  var s = String(x||"").trim();
  var m = s.match(/^(\d{1,2})\.(\d{1,2})\.(\d{4})/);
  if (!m) return new Date(0);
  return new Date(+m[3], +m[2]-1, +m[1]);
  }

  list.sort(function(a,b){
  if (sort === "tarih_desc")  return parseTRDate(b.tarih) - parseTRDate(a.tarih);
  if (sort === "tarih_asc")  return parseTRDate(a.tarih) - parseTRDate(b.tarih);
  if (sort === "toplam_desc") return parseTRMoney(b.genelToplam) - parseTRMoney(a.genelToplam);
  if (sort === "toplam_asc")  return parseTRMoney(a.genelToplam) - parseTRMoney(b.genelToplam);
  if (sort === "firma_desc")  return String(b.firma||"").localeCompare(String(a.firma||""), "tr");
  return String(a.firma||"").localeCompare(String(b.firma||""), "tr");
  });

  if (typeof renderTeklifGecmisi === "function") renderTeklifGecmisi(list);
}

function renderTeklifGecmisi(list) {
  var tb = document.getElementById("teklifBody");
  if (!tb) return;
  tb.innerHTML = "";
  if (!list || list.length === 0) {
    tb.innerHTML = "<tr><td colspan='9' style='text-align:center; padding:20px;'>Gösterilecek teklif bulunamadı.<\/td><\/tr>";
    return;
  }

  // Firma bazında grupla
  var gruplar = {};
  var firmaStr = [];
  list.forEach(function(r) {
    var key = String(r.firma || r.firmaAdi || "-").trim();
    if (!gruplar[key]) { gruplar[key] = []; firmaStr.push(key); }
    gruplar[key].push(r);
  });

  firmaStr.forEach(function(firmaAdi) {
    var satirlar = gruplar[firmaAdi];
    var birden = satirlar.length > 1;

    // Firma başlık satırı (birden fazla teklif varsa)
    if (birden) {

function raporKullaniciListesiniDoldur(liste) {
  var sel = document.getElementById("rapUser");
  if (!sel) return;
  
  // 1. Önce temizle ve varsayılanı ekle
  sel.innerHTML = '<option value="">Tümü<\/option>';
  
  // 2. ÖNEMLİ: Gelen verinin dizi olduğundan emin ol (Hata koruması)
  var asilListe = [];
  if (Array.isArray(liste)) {
    asilListe = liste;
  } else if (liste && Array.isArray(liste.liste)) {
    asilListe = liste.liste;
  } else if (window._personelCache && Array.isArray(window._personelCache)) {
    asilListe = window._personelCache;
  }

  // 3. Döngüye girmeden önce asilListe'nin dizi olduğunu bir kez daha garanti et
  if (!Array.isArray(asilListe)) return;

  // 4. Klasik FOR döngüsü (Hata riskini sıfıra indirir)
  for (var i = 0; i < asilListe.length; i++) {
    var p = asilListe[i];
    if (p && p.ad) {
      var opt = document.createElement("option");
      opt.value = p.mail || p.ad;
      opt.textContent = p.ad;
      sel.appendChild(opt);
    }
  }
}

function promoEsc(s) {
  return String(s || "").replace(/[&<>"']/g, function(m) {
  return {
  "&": "&amp;",
  "<": "&lt;",
  ">": "&gt;",
  '"': "&quot;",
  "'": "&#39;"
  }[m];
  });
}
function promoRenderTplList() {
  var box = document.getElementById("promoTplBox");
  if (!box) return;

  box.innerHTML = PROMO_TEMPLATES.map(function(t) {
  var checked = window.__promoState.selectedKeys.has(t.key) ? "checked" : "";
  return '<label style="display:flex; gap:10px; align-items:flex-start; padding:8px; border-bottom:1px solid #f1f5f9;">' +
  '<input type="checkbox" class="promoTplChk" data-k="' + t.key + '" ' + checked + ' style="margin-top:2px;">' +
  '<div>' +
  '<div style="font-weight:800;">' + promoEsc(t.title) + '<\/div>' +
  '<div class="muted" style="font-size:11px;">' + promoEsc(t.subject) + '<\/div>' +
  '<\/div>' +
  '<\/label>';
  }).join("");

  box.querySelectorAll(".promoTplChk").forEach(function(chk) {

  });
  });
}

function promoGuessSubject() {
  var firstKey = null;
  window.__promoState.selectedKeys.forEach(function(k) {
  if (!firstKey) firstKey = k;
  });
  var t = null;
  for (var i = 0; i < PROMO_TEMPLATES.length; i++) {
  if (PROMO_TEMPLATES[i].key === firstKey) { t = PROMO_TEMPLATES[i]; break; }
  }
  if (!t) {
  for (var j = 0; j < PROMO_TEMPLATES.length; j++) {
  if (PROMO_TEMPLATES[j].key === "GENEL") { t = PROMO_TEMPLATES[j]; break; }
  }
  }
  return t ? t.subject : "AvrupaISG - Tanitim";
}

function getFirmaInfoSmart_(firmaAdi, firmaId) {
  var list = Array.isArray(window._firmaCache) ? window._firmaCache : [];

  function normLocal(s) { return String(s || "").toLocaleLowerCase("tr-TR").trim(); }

  var o = null;
  if (firmaId) {
  for (var i = 0; i < list.length; i++) {
  if (String(list[i].id || list[i].firmaId || "").trim() === String(firmaId).trim()) {
  o = list[i]; break;
  }
  }
  }
  if (!o && firmaAdi) {
  for (var j = 0; j < list.length; j++) {
  if (normLocal(list[j].firma || list[j].unvan || list[j].firmaAdi || "") === normLocal(firmaAdi)) {
  o = list[j]; break;
  }
  }
  }

  var yetkili = o ? (o.yetkili || o.ilgili || o.yetkiliAdi || o.kisi || "") : "";
  var tel  = o ? (o.telefon || o.tel || o.cep || o.gsm || "") : "";
  var mail  = o ? (o.email || o.mail || o.eposta || "") : "";

  return {
  yetkili: String(yetkili || "").trim(),
  tel:  String(tel  || "").trim(),
  mail:  String(mail  || "").trim()
  };
}
