function mahsupRender(liste) {

    html += "<td style='padding:8px; text-align:right; color:#dc2626; font-weight:bold;'>" + Number(f.kalanTutar).toLocaleString("tr-TR") + " ₺<\/td>";

window.mahsupFiltrele = function() {
  var ara = (document.getElementById("mahsupAra").value||"").toLowerCase();
  var durum = document.getElementById("mahsupDurum").value;
  var filtreli = _mahsupFaturalar.filter(function(f) {
    var firmaUyum = !ara || f.firmaAdi.toLowerCase().indexOf(ara) > -1 || (f.faturaNo||'').toLowerCase().indexOf(ara) > -1;
    var durumUyum = !durum || f.durum === durum;
    return firmaUyum && durumUyum;
  });

};

      btn.style.background  = "white";

function gelirCariFirmaAra() {
  var ara = (document.getElementById("gelirCariFirmaAra").value || "").toLowerCase();
  var oneri = document.getElementById("gelirCariFirmaOneri");
  if (!ara || ara.length < 2) { oneri.style.display = "none"; return; }
  var firmalar = window._firmaCache || [];
  var eslesen = firmalar.filter(function(f) {
    return (f.FirmaAdi || "").toLowerCase().indexOf(ara) > -1;
  }).slice(0, 8);
  if (!eslesen.length) { oneri.style.display = "none"; return; }
  oneri.innerHTML = eslesen.map(function(f) {
    var ad = f.FirmaAdi || "";
    return '<div onclick="gelirCariFirmasec(\'' + ad.replace(/'/g,"\\'") + '\')" ' +
      'style="padding:10px 12px;cursor:pointer;border-bottom:1px solid #f1f5f9;font-size:13px;font-weight:600;" ' +
      'onmouseover="this.style.background=\'#f0f9ff\'" onmouseout="this.style.background=\'white\'">' +
      '🏢 ' + ad + '<\/div>';
  }).join("");
  oneri.style.display = "block";
}

function bankoGelirKaydet() {
  var tutar = parseFloat(document.getElementById("gelirTutar").value || 0);
  if (!tutar || tutar <= 0) { alert("Geçerli bir tutar giriniz!"); return; }

  var odemeTipi = document.getElementById("gelirOdemeTipi").value;
  var btn = document.querySelector("#bankoGelir button[onclick='bankoGelirKaydet()']");
  if (btn) { btn.disabled = true; btn.textContent = "Kaydediliyor..."; }

  // CARİ seçildiyse firma carisine ekle
  if (odemeTipi === "CARİ") {
    var cariFirmaId = document.getElementById("gelirCariFirmaId").value;
    var cariFirmaAdi = document.getElementById("gelirCariFirmaAdi").value;
    if (!cariFirmaId) { 
      if (btn) { btn.disabled = false; btn.textContent = "💾 GELİRİ KAYDET"; }
      alert("Lütfen firma seçiniz!"); 
      return; 
    }
    google.script.run
      .withSuccessHandler(function(res) {
        if (btn) { btn.disabled = false; btn.textContent = "💾 GELİRİ KAYDET"; }
        if (res.ok) {
          document.getElementById("gelirTutar").value = "";
          document.getElementById("gelirAciklama").value = "";
          document.getElementById("gelirCariFirmaAdi").value = "";
          document.getElementById("gelirCariFirmaId").value = "";
          document.getElementById("cariFirmaDiv").style.display = "none";
          bankoPanelYukle();
          bankoHastaSayisiGuncelle();
        } else { alert("❌ " + res.msg); }
      })
      .withFailureHandler(function(e) {
        if (btn) { btn.disabled = false; btn.textContent = "💾 GELİRİ KAYDET"; }
        alert("Hata: " + e);
      })
      .api_firmaCariEkle({
        firmaId: cariFirmaId,
        firmaAdi: cariFirmaAdi,
        tip: "ALACAK",
        tutar: tutar,
        aciklama: document.getElementById("gelirAciklama").value,
        kaydeden: window.currentUserEmail || ""
      });
    return;
  }

  // NAKİT, KREDİ KARTI, HAVALE/EFT
  var kasaAdi = odemeTipi === "HAVALE/EFT"
    ? document.getElementById("gelirBankaSecim").value
    : odemeTipi;
  var firmaAdi = document.getElementById("gelirFirmaAdi").value;

  google.script.run
    .withSuccessHandler(function(res) {
      if (btn) { btn.disabled = false; btn.textContent = "💾 GELİRİ KAYDET"; }
      if (res.ok) {
        document.getElementById("gelirTutar").value = "";
        document.getElementById("gelirAciklama").value = "";
        document.getElementById("gelirFirmaAdi").value = "";
        bankoPanelYukle();
        bankoHastaSayisiGuncelle();
      } else { alert("❌ " + res.msg); }
    })
    .withFailureHandler(function(e) {
      if (btn) { btn.disabled = false; btn.textContent = "💾 GELİRİ KAYDET"; }
      alert("Hata: " + e);
    })
    .api_gelirKaydet({
      tutar: tutar,
      odemeTipi: odemeTipi,
      kasa: kasaAdi,
      firmaAdi: firmaAdi,
      aciklama: document.getElementById("gelirAciklama").value,
      kaydedenEmail: window.currentUserEmail || ""
    });
}

function bankoWhatsappGonder() {
  var firmaAdi     = document.getElementById("bankoKargoFirmaAdi").value;
  var firmaIdEl    = document.getElementById("bankoKargoFirmaId");
  var firmaId      = firmaIdEl ? firmaIdEl.value : "";
  var takipNo      = document.getElementById("bankoTakipNo").value.trim();
  var kargoFirmasi = document.getElementById("bankoKargoFirması").value;
  var tel          = document.getElementById("bankoKargoTel").value.trim();

  if (!tel) { alert("Telefon numarası giriniz!"); return; }

  // Telefonu düzenle — başında 0 varsa 90 ekle
  tel = tel.replace(/\s+/g, "").replace(/^\+/, "");

function smEsc(s) {
  return String(s||'').split('&').join('&amp;').split('"').join('&quot;').split("'").join('&#39;').split('>').join('&gt;').split('<').join('&lt;');
}

function smLoadData() {
  google.script.run
    .withSuccessHandler(function(res) {
      if (res && res.ok) { SM_POSTS = res.liste || []; smRenderPosts(); smRenderKpis(); }
    })
    .withFailureHandler(function() { smRenderKpis(); })

}

function smRenderKpis() {
  var bugun = new Date().toISOString().split('T')[0];
  var gecikti=0, planlanan=0, paylasilan=0;
  for (var i=0; i < SM_POSTS.length; i++) {
    var p = SM_POSTS[i];
    if (p.durum==='paylasild') paylasilan++;
    if (p.durum==='planland' && p.tarih >= bugun) planlanan++;
    if (p.durum==='planland' && bugun > p.tarih) gecikti++;
  }
  smSet('sm-kpi1', SM_POSTS.length);
  smSet('sm-kpi2', paylasilan);
  smSet('sm-kpi3', planlanan);
  smSet('sm-kpi4', gecikti);
  var badge = document.getElementById('sm-gecikme-badge');
  if (badge) { badge.textContent = gecikti+' GECIKTI'; badge.style.display = gecikti>0?'inline':'none'; }
}

function isgZiyaretTarihleriniHesapla() {
  var sayi = parseInt(document.getElementById("iat_ziyaretSayisi").value) || 0;
  var ilk = document.getElementById("iat_ilkZiyaret").value;
  var onizleme = document.getElementById("iat_ziyaretOnizleme");
  if (!sayi || !ilk) { if(onizleme) onizleme.style.display="none"; return; }

  var tarihler = [];
  var bas = new Date(ilk);
  var aralikAy = Math.round(12 / sayi);

  for (var i = 0; i < sayi; i++) {
    var t = new Date(bas);
    t.setMonth(t.getMonth() + (i * aralikAy));
    var gun = String(t.getDate()).padStart(2,"0");
    var ay = String(t.getMonth()+1).padStart(2,"0");
    var yil = t.getFullYear();
    tarihler.push(gun + "." + ay + "." + yil);
  }

  window._isgZiyaretTarihleri = tarihler.join(",");

  if (onizleme) {
    onizleme.style.display = "block";
    onizleme.innerHTML = "<b style='color:#065f46;'>📅 Planlanan Ziyaret Tarihleri:<\/b><br>" +
      tarihler.map(function(t, i) {
        return "<span style='display:inline-block; background:#dcfce7; color:#065f46; padding:2px 8px; border-radius:4px; margin:3px; font-weight:bold;'>" + (i+1) + ". " + t + "<\/span>";
      }).join("");
 }
}

      if (!d || !d.ok || !d.liste || !d.liste.length) { div.innerHTML = '<div class="muh2-loading">Kayıt yok.<\/div>'; return; }
      var html = '<table class="muh2-tablo"><thead><tr><th>Tarih</th><th>Kullanıcı</th><th>İşlem</th><th>Detay</th></tr></thead><tbody>';
      d.liste.forEach(function(l) {
        var renk = l.islem.indexOf('SIL') > -1 ? '#fee2e2' : l.islem.indexOf('GELIR') > -1 ? '#f0fdf4' : '#f8fafc';
        html += '<tr style="background:' + renk + ';">'
          + '<td style="white-space:nowrap;font-size:11px;">' + (l.tarih || '-') + '<\/td>'
          + '<td style="font-size:11px;">' + (l.kullanici || '-').split('@')[0] + '<\/td>'
          + '<td><span style="background:#eff6ff;color:#1d4ed8;padding:2px 6px;border-radius:4px;font-size:10px;font-weight:700;">' + (l.islem || '-') + '<\/span></td>'
          + '<td style="font-size:11px;max-width:250px;overflow:hidden;text-overflow:ellipsis;white-space:nowrap;">' + (l.detay || '-') + '<\/td>'
          + '<\/tr>';
      });
      html += '<\/tbody><\/table>';
      div.innerHTML = html;
    })
    .withFailureHandler(function() { document.getElementById('muh2LogDiv').innerHTML = 'Hata.'; })
    .api_sonIslemLoglari(20);
}

function muh2MahsupKaydet() {
  var faturaId=document.getElementById('muh2MahsupFaturaId').value;
  var faturaNo=document.getElementById('muh2MahsupFaturaNo').value;
  var firmaAdi=document.getElementById('muh2MahsupFirmaAdi').textContent;

  var mahsupTutar=Number(document.getElementById('muh2MahsupTutarInput').value||0);
  var aciklama=document.getElementById('muh2MahsupAciklama').value;
  if(!mahsupTutar||mahsupTutar<=0){ alert('Mahsuplaşma tutarı giriniz!'); return; }
  google.script.run
    .withSuccessHandler(function(res){
      if(res&&res.ok){
        alert('✅ Mahsuplaşma kaydedildi!');
        document.getElementById('muh2MahsupForm').style.display='none';
        _muh2KasaBakiye+=mahsupTutar;
        muh2MahsupYukle();
        muh2KasaYukle();

    .withFailureHandler(function(e){ alert('❌ Hata: '+e.message); })
    .api_mahsupKaydet({faturaId:faturaId,faturaNo:faturaNo,firmaAdi:firmaAdi,faturaTutari:faturaTutar,mahsupTutari:mahsupTutar,aciklama:aciklama,kaydedenEmail:window.currentUserEmail||''});
