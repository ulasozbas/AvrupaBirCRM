function teklifOnayGonder(id, firma) {
  var hizmetler = [];
  if (document.getElementById('hizSecOSGB').checked) hizmetler.push('OSGB');
  if (document.getElementById('hizSecMobil').checked) hizmetler.push('MOBIL');
  if (document.getElementById('hizSecIseGiris').checked) hizmetler.push('ISE_GIRIS');

  if (!hizmetler.length) { alert('En az bir hizmet tipi seçin!'); return; }

  var modal = document.getElementById('hizmetSecModal');
  if (modal) modal.remove();

  google.script.run
    .withSuccessHandler(function(res) {
      if (res && res.ok) {
        if (typeof notify === "function") notify("✅ Teklif onaylandı ve atamalara gönderildi!", "success");
        if (typeof loadTeklifGecmisi === "function") loadTeklifGecmisi();
        if (typeof raporTeklifGetir === "function") {
          var f = (document.getElementById('rapBas')||{}).value;
          var t = (document.getElementById('rapBit')||{}).value;
          var e = (document.getElementById('rapEmail')||{}).value;
          raporTeklifGetir(f, t, e);
        }
      } else {
        alert("Hata: " + (res ? res.msg : "Güncellenemedi"));
      }
    })
    .withFailureHandler(function(err) { alert("Bağlantı hatası: " + err.message); })
    .api_teklifDurumGuncelle(id, "ONAYLANDI", hizmetler.join(","));
}

function mobilAtamaKontrol() {
  google.script.run
    .withSuccessHandler(function(d) {
      if (!d || !d.ok) return;
      var mobilSayi = 0, iseGirisSayi = 0;
      var mobilListe = [], iseGirisListe = [];
      (d.liste || []).forEach(function(a) {
        if (a.durum !== 'BEKLEMEDE') return;
        if (a.hizmetler === 'MOBIL') { mobilSayi++; mobilListe.push(a); }
        if (a.hizmetler === 'ISE_GIRIS') { iseGirisSayi++; iseGirisListe.push(a); }
      });
      var toplam = mobilSayi + iseGirisSayi;
      var uyari = document.getElementById('mobilAtamaUyari');
      if (uyari) uyari.style.display = toplam > 0 ? 'block' : 'none';
      var baslik = document.getElementById('mobilAtamaUyariBaslik');
      if (baslik) baslik.textContent = '🔔 ' + toplam + ' Bekleyen Atama!';
      var detay = document.getElementById('mobilAtamaUyariDetay');
      if (detay) detay.textContent = mobilSayi + ' Mobil Sağlık, ' + iseGirisSayi + ' İşe Giriş — detay için tıklayın';
      var mb = document.getElementById('mobilMobilBadge');
      if (mb) mb.textContent = '🚑 ' + mobilSayi;
      var ib = document.getElementById('mobilIseGirisBadge');
      if (ib) ib.textContent = '🏥 ' + iseGirisSayi;
      window._mobilAtamaListe = { mobil: mobilListe, iseGiris: iseGirisListe };
    })
    .withFailureHandler(function(){})
    .api_bekleyenAtamalar(window.currentUserEmail || "", "");
}

function mobilPlanlaAc(firmaAdi, firmaId, atamaId, atamaSatir, tip) {
  // Mobil planlama modalını aç
  var modal = document.getElementById('mobilGorevModal');
  if (modal) modal.style.display = 'flex';

  // Firma otomatik seç
  var selFirma = document.getElementById('mgFirmaId');
  if (selFirma) {
    for (var i = 0; i < selFirma.options.length; i++) {
      if (selFirma.options[i].value === firmaId || selFirma.options[i].text.toUpperCase().indexOf(firmaAdi.toUpperCase()) > -1) {
        selFirma.selectedIndex = i; break;
      }
    }
  }

  // Arama kutusuna firma adını yaz
  var araEl = document.getElementById('mgFirmaAra');
  if (araEl) araEl.value = firmaAdi;

  // Durum ayarla
  var durumEl = document.getElementById('mgDurum');
  if (durumEl) durumEl.value = 'YAPILACAK';

  // Atama bilgisini sakla
  window._mobilAtamaKayit = { atamaId: atamaId, atamaSatir: atamaSatir, tip: tip };

  // Bugünün tarihini set et
  var tarihEl = document.getElementById('mgTarih');
  if (tarihEl && !tarihEl.value) tarihEl.value = new Date().toISOString().split('T')[0];

  // Mobil ekip listesine otomatik bir personel satırı ekle
  var ekipDiv = document.getElementById('mgEkipListesi');
  if (ekipDiv) {
    ekipDiv.innerHTML = '';
    mgEkipEkle();
  }
}

  var renk = Math.abs(fark) < 1 ? '#10b981' : '#ef4444';
  var mesaj = Math.abs(fark) < 1 ? '✅ Eşleşiyor' : (fark > 0 ? '⬆️ Fazla: ₺' + fark.toLocaleString('tr-TR') : '⬇️ Eksik: ₺' + Math.abs(fark).toLocaleString('tr-TR'));

function isgKapasiteExcelImport(input) {
  var file = input.files[0];
  if (!file) return;
  var reader = new FileReader();
  reader.onload = function(e) {
    var data = new Uint8Array(e.target.result);
    var workbook = XLSX.read(data, { type: 'array' });
    var sheet = workbook.Sheets[workbook.SheetNames[0]];
    var json = XLSX.utils.sheet_to_json(sheet, { header: 1 });
    if (json.length < 2) { alert('Excel boş!'); return; }

    var satirlar = [];
    for (var i = 1; i < json.length; i++) {
      var r = json[i];
      if (!r[0]) continue;
      var sertifika = String(r[1] || '');
      var sinif = 'C';
      if (sertifika.indexOf('A Sınıfı') > -1) sinif = 'A';
      else if (sertifika.indexOf('B Sınıfı') > -1) sinif = 'B';
      else if (sertifika.indexOf('C Sınıfı') > -1) sinif = 'C';
      else if (sertifika.indexOf('Hekim') > -1) sinif = 'Hekim';
      satirlar.push([
        String(r[0] || '').trim(),
        sertifika.trim(),
        sinif,
        Number(r[2]) || 0,
        String(r[3] || '').trim(),
        String(r[4] || '').trim(),
        Number(r[5]) || 0,
        String(r[6] || '').trim(),
        String(r[7] || '')
      ]);
    }
    if (satirlar.length === 0) { alert('Geçerli satır bulunamadı!'); return; }
    if (!confirm(satirlar.length + ' kayıt import edilecek. Devam?')) return;

    google.script.run
      .withSuccessHandler(function(res) {
        if (res.ok) {
          alert('✅ ' + res.msg);
          isgKapasiteDashYukle();
        } else {
          alert('❌ ' + res.msg);
        }
      })
      .withFailureHandler(function(err) { alert('Hata: ' + err.message); })
      .api_isgKapasiteImport(satirlar);
  };
  reader.readAsArrayBuffer(file);
  input.value = '';
}

function isgSinifBelirle(adSoyad, unvan) {
  if (!adSoyad) return '';
  if (_isgSinifMap[adSoyad]) return _isgSinifMap[adSoyad];
  var aranan = adSoyad.toUpperCase().trim();
  for (var k in _isgSinifMap) {
    if (k.toUpperCase().trim() === aranan) return _isgSinifMap[k];
  }
  if (unvan) {
    var u = String(unvan);
    if (u.indexOf('A Sınıfı') > -1) return 'A';
    if (u.indexOf('B Sınıfı') > -1) return 'B';
    if (u.indexOf('C Sınıfı') > -1) return 'C';
    if (u.indexOf('Hekim') > -1 || u.indexOf('Doktor') > -1) return 'Hekim';
  }
  return '';
}

function isgSinifUygunMu(personelSinif, tehlikeSinifi) {
  if (!personelSinif || !tehlikeSinifi) return true;
  if (personelSinif === 'Hekim' || personelSinif === 'A') return true;
  if (personelSinif === 'B') return tehlikeSinifi !== 'Çok Tehlikeli';
  if (personelSinif === 'C') return tehlikeSinifi === 'Az Tehlikeli';
  return true;
}

        h += '<\/div>';
      }
      if (o.hekimOneri && o.hekimOneri.ad) {
        h += '<div style="background:#fff;border-radius:8px;padding:10px;margin-bottom:8px;border-left:3px solid #1d4ed8;">';
        h += '<div style="font-weight:700;color:#1d4ed8;font-size:13px;">🩺 Hekim: ' + o.hekimOneri.ad + '<\/div>';
        h += '<div style="font-size:12px;color:#475569;margin-top:2px;">' + o.hekimOneri.sebep + '<\/div>';

        h += '<\/div>';

    })

var _isgCografiData = null;
var _isgCografiMap = null;
var _isgCografiMarkers = null;

var ISG_PER_RENKLER = ['#ef4444','#3b82f6','#22c55e','#f59e0b','#8b5cf6','#ec4899','#14b8a6','#f97316','#6366f1','#84cc16','#06b6d4','#e11d48'];

function isgCografiYukle() {
  document.getElementById('isgCografiKPI').innerHTML = '<div style="text-align:center;padding:20px;color:#64748b;width:100%;">Yükleniyor...<\/div>';
  document.getElementById('isgCografiSidebar').innerHTML = '';
  google.script.run
    .withSuccessHandler(function(r) {
      if (!r || !r.ok) {
        document.getElementById('isgCografiKPI').innerHTML = '<div style="color:#ef4444;padding:20px;">❌ ' + (r ? r.msg : 'Hata') + '<\/div>';
        return;
      }
      _isgCografiData = r;
      isgCografiRender();
    })
    .withFailureHandler(function(e) {
      document.getElementById('isgCografiKPI').innerHTML = '<div style="color:#ef4444;padding:20px;">Hata: ' + e.message + '<\/div>';
    })
    .api_isgCografiVeri();
}

function isgCografiRender() {
  var d = _isgCografiData;
  if (!d) return;
  var o = d.ozet;

  // KPI
  var kh = '';
  kh += isgKapaKPI('🗺️', 'GPS Var', o.gpsVar, o.toplamFirma + ' firmadan', '#22c55e');
  kh += isgKapaKPI('📍', 'GPS Yok', o.gpsYok, 'Konum girilmeli', '#ef4444');
  kh += isgKapaKPI('👥', 'Personel', o.personelSayisi, 'Haritada gösteriliyor', '#3b82f6');
  document.getElementById('isgCografiKPI').innerHTML = kh;

  // Harita
  var mapEl = document.getElementById('isgCografiHarita');
  if (_isgCografiMap) { _isgCografiMap.remove(); _isgCografiMap = null; }
  _isgCografiMap = L.map(mapEl, { center: [41.015, 28.979], zoom: 11 });
  L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', { maxZoom: 19 }).addTo(_isgCografiMap);
  _isgCografiMarkers = L.layerGroup().addTo(_isgCografiMap);

  // Personel renk haritası
  var perRenk = {};
  var rIdx = 0;
  for (var p = 0; p < d.personeller.length; p++) {
    perRenk[d.personeller[p].ad] = ISG_PER_RENKLER[rIdx % ISG_PER_RENKLER.length];
    rIdx++;
  }

  // Marker ekle
  var bounds = [];
  for (var i = 0; i < d.firmalar.length; i++) {
    var f = d.firmalar[i];
    var renk = perRenk[f.personel] || '#64748b';
    var tehRenk = f.tehlike === 'Çok Tehlikeli' ? '#ef4444' : f.tehlike === 'Tehlikeli' ? '#f59e0b' : '#22c55e';
    var icon = L.divIcon({
      className: '',
      html: '<div style="width:14px;height:14px;border-radius:50%;background:' + renk + ';border:2px solid #fff;box-shadow:0 1px 4px rgba(0,0,0,0.3);"><\/div>',
      iconSize: [14, 14],
      iconAnchor: [7, 7]
    });
    var marker = L.marker([f.enlem, f.boylam], { icon: icon }).addTo(_isgCografiMarkers);
    marker.bindPopup(
      '<div style="font-size:12px;min-width:180px;">' +
      '<div style="font-weight:700;margin-bottom:4px;">' + f.ad + '<\/div>' +
      '<div style="color:#64748b;">📍 ' + (f.adres || 'Adres yok') + '<\/div>' +
      '<div style="margin-top:4px;"><span style="background:' + tehRenk + ';color:#fff;padding:1px 6px;border-radius:4px;font-size:10px;">' + f.tehlike + '<\/span><\/div>' +
      '<div style="margin-top:4px;">🦺 ' + (f.personel || '-') + ' (' + f.iguDk + ' dk)<\/div>' +
      '<div>🩺 ' + (f.hekim || '-') + ' (' + f.hekimDk + ' dk)<\/div>' +
      '<\/div>'
    );
    bounds.push([f.enlem, f.boylam]);
  }
  if (bounds.length > 0) _isgCografiMap.fitBounds(bounds, { padding: [30, 30] });

  // Sidebar — personel legend
  var sh = '<div style="font-weight:700;font-size:14px;margin-bottom:12px;color:#1e293b;">👥 Personel Bölgeleri<\/div>';
  for (var j = 0; j < d.personeller.length; j++) {
    var pr = d.personeller[j];
    var renk2 = perRenk[pr.ad] || '#64748b';
    sh += '<div onclick="isgCografiFiltre(\'' + pr.ad.replace(/'/g, "\\'") + '\')" style="background:#fff;border:1px solid #e2e8f0;border-left:4px solid ' + renk2 + ';border-radius:8px;padding:10px;margin-bottom:8px;cursor:pointer;">';
    sh += '<div style="font-weight:700;font-size:13px;color:#1e293b;">' + pr.ad + '<\/div>';
    sh += '<div style="font-size:11px;color:#64748b;margin-top:2px;">' + pr.firmaSayisi + ' firma · ' + pr.gpsli + ' GPS · ' + pr.topDk + ' dk<\/div>';
    sh += '<\/div>';
  }
  document.getElementById('isgCografiSidebar').innerHTML = sh;

  setTimeout(function() { _isgCografiMap.invalidateSize(); }, 200);
}

function isgCografiFiltre(personelAd) {
  if (!_isgCografiData || !_isgCografiMap) return;
  _isgCografiMarkers.clearLayers();
  var firmalar = personelAd === 'all' ? _isgCografiData.firmalar : _isgCografiData.firmalar.filter(function(f) { return f.personel === personelAd; });
  var perRenk = {};
  var rIdx = 0;
  for (var p = 0; p < _isgCografiData.personeller.length; p++) {
    perRenk[_isgCografiData.personeller[p].ad] = ISG_PER_RENKLER[rIdx % ISG_PER_RENKLER.length];
    rIdx++;
  }
  var bounds = [];
  for (var i = 0; i < firmalar.length; i++) {
    var f = firmalar[i];
    var renk = perRenk[f.personel] || '#64748b';
    var icon = L.divIcon({
      className: '',
      html: '<div style="width:14px;height:14px;border-radius:50%;background:' + renk + ';border:2px solid #fff;box-shadow:0 1px 4px rgba(0,0,0,0.3);"><\/div>',
      iconSize: [14, 14],
      iconAnchor: [7, 7]
    });
    L.marker([f.enlem, f.boylam], { icon: icon }).addTo(_isgCografiMarkers);
    bounds.push([f.enlem, f.boylam]);
  }
  if (bounds.length > 0) _isgCografiMap.fitBounds(bounds, { padding: [30, 30] });
}

function isgSimRender(r, firma, dk, ay) {
  var o = r.ozet;
  var h = '';
  h += '<div style="display:flex;gap:12px;flex-wrap:wrap;margin-bottom:16px;">';
  h += isgKapaKPI('🔮', ay + ' Ay Sonra Kritik', o.kritikSayisi + ' kişi', 'Doluluk ≥%95 olacak', '#ef4444');
  h += isgKapaKPI('🏢', 'Alabileceğiniz Firma', o.topAlabilecekFirma, 'Ort. ' + dk + ' dk/firma ile', '#22c55e');
  h += isgKapaKPI('⏱️', 'Toplam Boş Kapasite', o.topKalanDk.toLocaleString('tr-TR') + ' dk', o.personelSayisi + ' personel', '#3b82f6');
  h += '<\/div>';

  h += '<div style="display:grid;grid-template-columns:repeat(auto-fill,minmax(280px,1fr));gap:10px;">';
  for (var i = 0; i < r.liste.length; i++) {
    var p = r.liste[i];
    var renk = p.sonPct >= 95 ? '#ef4444' : p.sonPct >= 80 ? '#f59e0b' : '#22c55e';
    var bgRenk = p.sonPct >= 95 ? '#fef2f2' : p.sonPct >= 80 ? '#fffbeb' : '#f0fdf4';

    h += '<div style="background:#fff;border:1px solid #e2e8f0;border-radius:10px;overflow:hidden;">';
    h += '<div style="background:' + bgRenk + ';padding:10px 14px;border-bottom:1px solid #e2e8f0;">';
    h += '<div style="display:flex;justify-content:space-between;align-items:center;">';
    h += '<span style="font-weight:700;font-size:13px;">' + p.ad + '<\/span>';
    h += '<span style="font-size:11px;color:#64748b;">' + p.sinif + '<\/span>';
    h += '<\/div><\/div>';

    h += '<div style="padding:10px 14px;">';
    // Mevcut → Gelecek bar
    h += '<div style="display:flex;justify-content:space-between;font-size:11px;margin-bottom:4px;">';
    h += '<span>Şimdi: <strong>%' + p.mevcutPct + '<\/strong><\/span>';
    h += '<span style="color:' + renk + ';">' + ay + ' ay sonra: <strong>%' + p.sonPct + '<\/strong><\/span>';
    h += '<\/div>';
    h += '<div style="position:relative;height:16px;background:#f1f5f9;border-radius:6px;overflow:hidden;">';
    h += '<div style="width:' + Math.min(p.mevcutPct, 100) + '%;height:100%;background:#94a3b8;border-radius:6px;position:absolute;"><\/div>';
    h += '<div style="width:' + Math.min(p.sonPct, 100) + '%;height:100%;background:' + renk + '55;border-radius:6px;position:absolute;border-right:2px solid ' + renk + ';"><\/div>';
    h += '<\/div>';

    // Detay
    h += '<div style="margin-top:8px;font-size:11px;color:#64748b;">';
    if (p.dolacakAy > -1) {
      h += '<div style="color:#ef4444;font-weight:700;">⚠ ' + p.dolacakAy + '. ayda dolacak!<\/div>';
    }
    h += '<div>Kalan: ' + p.kalan.toLocaleString('tr-TR') + ' dk · Doldurmak için ~' + p.doldurmaFirma + ' firma<\/div>';
    h += '<\/div>';

    h += '<\/div><\/div>';
  }
  h += '<\/div>';

  document.getElementById('simSonucDiv').innerHTML = h;
}

function devirTalepKontrol() {
  google.script.run
    .withSuccessHandler(function(liste) {
      if (!liste) liste = [];
      var btn = document.getElementById('devirBildirimBtn');
      var badge = document.getElementById('devirBadge');
      if (liste.length > 0) {
        btn.style.display = 'flex';
        badge.textContent = liste.length;
        window._devirTalepler = liste;
      } else {
        btn.style.display = 'none';
      }
    })
    .withFailureHandler(function() {})
    .api_bekleyenAtamaTalepleri();
}

function devirTalepPanelAc() {
  var panel = document.getElementById('devirTalepPanel');
  panel.style.display = panel.style.display === 'none' ? 'block' : 'none';
  if (panel.style.display === 'block') devirTalepListeGoster();
}

function devirTalepListeGoster() {
  var liste = window._devirTalepler || [];
  var el = document.getElementById('devirTalepListesi');
  if (liste.length === 0) {
    el.innerHTML = '<div style="text-align:center;padding:20px;color:#10b981;font-weight:700;">✅ Bekleyen talep yok.<\/div>';
    return;
  }
  var h = '';
  liste.forEach(function(t) {
    h += '<div style="background:#faf5ff;border:1px solid #e9d5ff;border-radius:10px;padding:12px;margin-bottom:8px;">';
    h += '<div style="font-weight:700;font-size:13px;color:#1e293b;">' + (t.firmaAdi || '-') + '<\/div>';
    h += '<div style="font-size:11px;color:#64748b;margin-top:4px;">👤 ' + (t.isteyenAdSoyad || t.isteyenEmail || '-') + '<\/div>';
    h += '<div style="font-size:11px;color:#94a3b8;margin-top:2px;">📅 ' + (t.tarih || '-') + '<\/div>';
    h += '<div style="display:flex;gap:6px;margin-top:8px;">';
    h += '<button onclick="devirTalepOnayla(\'' + t.firmaId + '\')" style="flex:1;background:#065f46;color:#fff;border:none;padding:7px;border-radius:6px;font-weight:700;cursor:pointer;font-size:11px;">✅ Onayla<\/button>';
    h += '<button onclick="devirTalepReddet(\'' + t.firmaId + '\')" style="flex:1;background:#ef4444;color:#fff;border:none;padding:7px;border-radius:6px;font-weight:700;cursor:pointer;font-size:11px;">❌ Reddet<\/button>';
    h += '<\/div><\/div>';
  });
  el.innerHTML = h;
}

function devirTalepOnayla(firmaId) {
  if (!confirm('Bu devir talebini onaylıyor musunuz?')) return;
  google.script.run
    .withSuccessHandler(function(r) {
      if (r && r.ok) {
        notify('✅ Devir onaylandı!', 'success');
        devirTalepKontrol();
        devirTalepListeGoster();
      } else {
        alert('❌ ' + (r ? r.msg : 'Hata'));
      }
    })
    .withFailureHandler(function(e) { alert('❌ ' + e.message); })
    .api_firmaAtaOnay({ firmaId: firmaId, adminEmail: window.currentUserEmail || '' });
}

function devirTalepReddet(firmaId) {
  if (!confirm('Bu devir talebini reddetmek istiyor musunuz?')) return;
  google.script.run
    .withSuccessHandler(function(r) {
      if (r && r.ok) {
        notify('✅ Talep reddedildi.', 'success');
        devirTalepKontrol();
        devirTalepListeGoster();
      } else {
        alert('❌ ' + (r ? r.msg : 'Hata'));
      }
    })
    .withFailureHandler(function(e) { alert('❌ ' + e.message); })
    .api_firmaAtaRed({ firmaId: firmaId, adminEmail: window.currentUserEmail || '' });
}
