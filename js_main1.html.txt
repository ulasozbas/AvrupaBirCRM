// Beni Hatırla - sayfa açılınca kontrol et
(function() {
  var kayitliEmail = localStorage.getItem("crm_email");
  var kayitliPin   = localStorage.getItem("crm_pin");
  if (kayitliEmail && kayitliPin) {
    var emailEl = document.getElementById("pinEmail");
    var pinEl   = document.getElementById("pinCode");
    var hatirlaEl = document.getElementById("beniHatirla");
    if (emailEl) emailEl.value = kayitliEmail;
    if (pinEl)   pinEl.value   = kayitliPin;
    if (hatirlaEl) hatirlaEl.checked = true;
  }
})();

var _origDoLogin = window.doLogin;
window.doLogin = function(e) {
  var hatirla = (document.getElementById("beniHatirla")||{}).checked;
  var email   = (document.getElementById("pinEmail")||{}).value || "";
  var pin     = (document.getElementById("pinCode")||{}).value || "";
  if (hatirla) {
    localStorage.setItem("crm_email", email);
    localStorage.setItem("crm_pin", pin);
  } else {
    localStorage.removeItem("crm_email");
    localStorage.removeItem("crm_pin");
  }
  _origDoLogin(e);
};

function teklifRevizeBaslat(asilTeklifNo) {
  var yeniNo = asilTeklifNo;
  var rMatch = asilTeklifNo.match(/-R(\d+)$/);
  if (rMatch) {
    yeniNo = asilTeklifNo.replace(/-R\d+$/, "-R" + (parseInt(rMatch[1]) + 1));
  } else {
    yeniNo = asilTeklifNo + "-R1";
  }

  google.script.run
    .withSuccessHandler(function(satirlar) {
      google.script.run
        .withSuccessHandler(function(list) {
          var r = null;
          if (list) {
            for (var i = 0; i < list.length; i++) {
              if (String(list[i].teklifNo||"").trim() === String(asilTeklifNo).trim()) {
                r = list[i]; break;
              }
            }
          }
          if (typeof showPanel === "function") showPanel("panelTeklif");
          setTimeout(function() {
            // Teklif no
            var noEl = document.getElementById("teklifNo");
            if (noEl) noEl.value = yeniNo;

            // Firma bilgileri
            if (r) {
              var firmaEl = document.getElementById("firma");
              if (firmaEl) firmaEl.value = r.firma || "";
              var yetkiliEl = document.getElementById("yetkili");
              if (yetkiliEl) yetkiliEl.value = r.yetkili || "";
              var mailEl = document.getElementById("mail");
              if (mailEl) mailEl.value = r.mailTo || "";
            }

            // Asıl teklif no gizli alan
            var asilEl = document.getElementById("asilTeklifNo");
            if (!asilEl) {
              asilEl = document.createElement("input");
              asilEl.type = "hidden";
              asilEl.id = "asilTeklifNo";
              document.body.appendChild(asilEl);
            }
            asilEl.value = asilTeklifNo;

            // Kalemleri forma yükle
            if (satirlar && satirlar.length > 0 && typeof satirEkleForm === "function") {
              satirlar.forEach(function(s) {
                satirEkleForm(s.grup, s.ad, s.adet, s.birim, s.kdv, s.maliyet);
              });
              alert("✏️ Revize modu: " + yeniNo + "\n" + satirlar.length + " kalem yüklendi. Düzenleyip kaydedin.");
            } else {
              alert("✏️ Revize modu: " + yeniNo + "\nFirma bilgileri yüklendi. Kalemleri ekleyin.");
            }
          }, 500);
        })
        .withFailureHandler(function(err) { alert("Hata: " + err.message); })
        .api_teklifGecmisi();
    })
    .withFailureHandler(function(err) { alert("Hata: " + err.message); })
    .api_teklifSatirlariniGetir(asilTeklifNo);
}

function pinDegistirKaydet(email) {
  var p1 = document.getElementById("yeniPin1").value.trim();
  var p2 = document.getElementById("yeniPin2").value.trim();
  var msg = document.getElementById("pinDegMsg");
  if (!p1 || p1.length < 4) { msg.textContent = "PIN en az 4 karakter olmalı."; return; }
  if (p1 !== p2) { msg.textContent = "PIN'ler eşleşmiyor."; return; }
  if (p1 === "1234") { msg.textContent = "1234 kullanamazsın."; return; }
  msg.textContent = "Kaydediliyor...";
  google.script.run
    .withSuccessHandler(function(r) {
      if (r && r.ok) {
        var m = document.getElementById("pinDegistirModal");
        if (m) m.remove();
        var res = window._pinDegRes;
        res.pinDegistir = false;
        if (typeof doLoginSonrasi === "function") doLoginSonrasi(res);
      } else {
        msg.textContent = r ? r.msg : "Hata oluştu.";
      }
    })
    .withFailureHandler(function(e) { msg.textContent = "Sunucu hatası: " + e.message; })
    .api_pinDegistir(email, p1);
}

function takipSecilenlereMail() {
  var secilen = [];
  document.querySelectorAll(".takipSec:checked").forEach(function(cb) {
    secilen.push({ firmaId: cb.dataset.id, firmaAdi: cb.dataset.firma, mail: cb.dataset.mail });
  });
  if (secilen.length === 0) { alert("Lütfen en az bir firma seç."); return; }
  if (!confirm(secilen.length + " firmaya tanıtım maili gönderilecek. Onaylıyor musun?")) return;
  var basarili = 0, hatali = 0, toplam = secilen.length;
  var bilgi = document.getElementById("takipMailBilgi");
  if (bilgi) bilgi.textContent = "Gönderiliyor... 0 / " + toplam;
  function gonder(idx) {
    if (idx >= secilen.length) {
      if (bilgi) bilgi.textContent = toplam + " firmadan " + basarili + " tanesi gönderildi, " + hatali + " hata.";
      takipPanelYukle();
      return;
    }
    var f = secilen[idx];
    google.script.run
      .withSuccessHandler(function(r) {
        if (r && r.ok) {
          basarili++;
          google.script.run.api_takipGuncelle(f.firmaId, "Tanıtım maili gönderildi", "BEKLIYOR", window.aktifKullanici);
        } else { hatali++; }
        if (bilgi) bilgi.textContent = "Gönderiliyor... " + (idx+1) + " / " + toplam + " — " + f.firmaAdi + (r&&r.ok?" ✅":" ❌");
        gonder(idx + 1);
      })
      .withFailureHandler(function(err) {
        hatali++;
        if (bilgi) bilgi.textContent = "HATA: " + (err && err.message ? err.message : JSON.stringify(err));
        gonder(idx + 1);
      })
      .api_tanitimMailGonder({
        senderEmail:   window.aktifKullanici,
        senderAdSoyad: window.currentUserAdSoyad || "",
        senderTel: window.currentUserTel || "",
        firmaId:       f.firmaId,
        firma:         f.firmaAdi,
        to:            f.mail
      });
  }
  gonder(0);
}
function izinTalepGonderUI() {
  var bas = document.getElementById("izinBaslangic").value;
  var bit = document.getElementById("izinBitis").value;
  var aciklama = document.getElementById("izinAciklama").value.trim();
  
  if (!bas || !bit || !aciklama) {
  notify("⚠️ Başlangıç, bitiş ve gerekçe zorunlu!", "error");
  return;
  }

  var d1 = new Date(bas), d2 = new Date(bit);
  var gun = Math.round((d2 - d1) / (1000 * 60 * 60 * 24)) + 1;
  if (gun <= 0) { notify("⚠️ Bitiş tarihi başlangıçtan önce olamaz!", "error"); return; }

  var btn = document.getElementById("btnIzinGonder");
  btn.disabled = true; btn.innerHTML = "Gönderiliyor... ⏳";

  .withSuccessHandler(function(res) {
  btn.disabled = false; btn.innerHTML = "🚀 TALEBİ GÖNDER";

  notify("✅ İzin talebiniz gönderildi!", "success");
  document.getElementById("izinBaslangic").value = "";
  document.getElementById("izinBitis").value = "";
  document.getElementById("izinAciklama").value = "";

  } else {
  alert("❌ " + (res ? res.msg : "Hata"));
  }
  })
  .withFailureHandler(function(err) {
  btn.disabled = false; btn.innerHTML = "🚀 TALEBİ GÖNDER";
  alert("❌ " + err.message);
  })
  .api_ikTalepKaydet({
  tur: "İZİN",
  miktar: gun,
  baslangic: bas,
  bitis: bit,
  aciklama: aciklama,
  ilgiliPersonel: window.currentUserEmail,
  talepEden: window.currentUserEmail,

// Tabloyu ekrana basan ana ve TEK fonksiyon
function personelTabloRender(gelenVeri) {
  var tbody = document.getElementById("personelListesiBody");
  if (!tbody) return;

  // 1. Veriyi diziye zorla (Hata koruması)
  var asilListe = [];
  if (Array.isArray(gelenVeri)) {
    asilListe = gelenVeri;
  } else if (gelenVeri && Array.isArray(gelenVeri.liste)) {
    asilListe = gelenVeri.liste;
  } else if (window._personelCache && Array.isArray(window._personelCache)) {
    asilListe = window._personelCache;
  }

  // 2. Filtre değerlerini al
  var fDurum = ((document.getElementById("personelDurumFiltre") || {}).value || "").toUpperCase().trim();
  var fAra = ((document.getElementById("personelAraInput") || {}).value || "").toLowerCase().trim();

  var html = "";
  var sayac = 0;

  // 3. Klasik FOR döngüsü (En güvenli yol)
  for (var i = 0; i < asilListe.length; i++) {
    var p = asilListe[i];
    
    // Filtreleme Mantığı
    var durum = String(p.durum || "").toUpperCase().trim();
    if (fDurum && durum !== fDurum) continue;
    
    var araMetin = [p.ad, p.mail, p.unvan].join(" ").toLowerCase();
    if (fAra && araMetin.indexOf(fAra) === -1) continue;

    sayac++;
    var renk = durum === "AKTİF" ? "color:#166534; background:#dcfce7;" : "color:#991b1b; background:#fee2e2;";
    
    html += '<tr style="border-bottom:1px solid #eee; cursor:pointer;" onclick="personelKartGoster(' + JSON.stringify(p).replace(/"/g, '&quot;') + ')">';
html += '<td style="padding:12px; font-weight:bold;">' + (p.ad || "-") + '<\/td>';
html += '<td style="padding:12px; color:#64748b;">' + (p.mail || "-") + '<\/td>';
html += '<td style="padding:12px; text-align:center; font-weight:bold; color:#0984e3;">' + (p.islemSayisi || 0) + '<\/td>';
html += '<td style="padding:12px;">' + (p.unvan || "-") + '<\/td>';
html += '<td style="padding:12px; font-weight:bold;">' + Number(p.maas || 0).toLocaleString("tr-TR") + ' ₺<\/td>';
html += '<td style="padding:12px; color:#2563eb; font-weight:bold;">' + (p.kalanIzin || 0) + ' Gün<\/td>';
html += '<td style="padding:12px;"><span style="' + renk + ' padding:4px 10px; border-radius:20px; font-size:11px;">' + durum + '<\/span><\/td>';
html += '<td style="padding:12px; text-align:center;"><button type="button" onclick="event.stopPropagation(); adminPersonelIslemAc(\'' + (p.id || "") + '\',\'' + (p.ad || "") + '\',\'' + (p.mail || "") + '\')" style="background:#1e293b; color:white; border:none; padding:5px 10px; border-radius:6px; font-size:11px; font-weight:bold; cursor:pointer;">İşlem<\/button><\/td>';
html += '<\/tr>';
}

tbody.innerHTML = html || '<tr><td colspan="8" style="text-align:center; padding:20px;">Kayıt bulunamadı.<\/td><\/tr>';
}

function recalcTable(){
const rows = Array.from(document.querySelectorAll("#tblBody tr"));

let net10 = 0, net20 = 0, netOther = 0;
let maliyet = 0;

for (const tr of rows){
  const rAdet = tr.querySelector(".r_adet") ? tr.querySelector(".r_adet").value : 0;
  const rBirim = tr.querySelector(".r_birim") ? tr.querySelector(".r_birim").value : 0;
  const rMaliyet = tr.querySelector(".r_maliyet") ? tr.querySelector(".r_maliyet").value : 0;
  const rKdv = tr.querySelector(".r_kdv") ? tr.querySelector(".r_kdv").value : 10;

  const adet = Number(rAdet) || 0;
  const birim = Number(rBirim) || 0;
  const m = Number(rMaliyet) || 0;
  const kdvOran = Number(rKdv) || 10; 

  const satirNet = adet * birim;

  if (kdvOran === 20) {
  net20 += satirNet;
  } else {
  net10 += satirNet; 
  }

  maliyet += adet * m;
}

const totalNet = net10 + net20 + netOther;

const iskInp = String(document.getElementById("iskonto")?.value || "0").replace(",", ".");
let isk = parseFloat(iskInp) || 0;
if (!isFinite(isk) || isk <= 0) isk = 0;
if (isk > totalNet) isk = totalNet;


const ratio10 = totalNet > 0 ? (net10 / totalNet) : 0;
const ratio20 = totalNet > 0 ? (net20 / totalNet) : 0;
const ratioO  = totalNet > 0 ? (netOther / totalNet) : 0;

const net10_after = Math.max(0, net10 - isk * ratio10);
const net20_after = Math.max(0, net20 - isk * ratio20);
const netO_after  = Math.max(0, netOther - isk * ratioO);

const vat10 = net10_after * 0.10;
const vat20 = net20_after * 0.20;

const totalVat = vat10 + vat20; 

const netAfterDiscount = Math.max(0, totalNet - isk);

const kdvDahil = document.getElementById("kdvDahil")?.checked !== false;

const genelToplam = kdvDahil
  ? (netAfterDiscount + totalVat)
  : netAfterDiscount;

const ozet = document.getElementById("tblOzet");
if (ozet){
  ozet.innerHTML = "";
  let html = "";

  html += "Satış: <b>" + money(totalNet) + " ₺<\/b><br>";

if (isk > 0.0001){
  html += "İskonto: <b>-" + money(isk) + " ₺<\/b><br>";
  }

  html += "KDV %10: <b>" + money(vat10) + " ₺<\/b><br>";
  html += "KDV %20: <b>" + money(vat20) + " ₺<\/b><br>";

  html += "<div style='margin-top:6px;border-top:1px solid #e5e7eb;padding-top:6px;'>" +
  "GENEL TOPLAM: <b>" + money(genelToplam) + " ₺<\/b>" +
  "<\/div>";

  html += "<div class='muted' style='margin-top:6px;'>" +
  (kdvDahil ? "Toplamlar KDV dahildir." : "Toplamlar KDV hariçtir.") +
  "<\/div>";

  ozet.innerHTML = html;
}
}

  var mesafeArtis = mesafe > 10 ? 10 : 0;

  // Vardiya artışı
  var vardiyaArtis = vardiya > 1 ? 10 : 0;

  // Net oran (+ artış, - indirim)
  var netOran = mesafeArtis + vardiyaArtis - kisiIndirim;

  window._mfp_oran = netOran;
  window._mfp_fiyat = netOran; // artık fiyat değil oran tutuyoruz

  var sonucDiv  = document.getElementById("mfp_sonuc");
  var fiyatDiv  = document.getElementById("mfp_fiyat");
  var detayDiv  = document.getElementById("mfp_detay");
  var uygulaBtn = document.getElementById("mfp_uygulaBtn");

  var oranYazi = (netOran >= 0 ? "+" : "") + netOran + "%";
  fiyatDiv.textContent = oranYazi;
  fiyatDiv.style.color = netOran > 0 ? "#b91c1c" : netOran < 0 ? "#15803d" : "#1e293b";

  detayDiv.textContent =
    (kisiIndirim  ? "Kişi indirimi: −" + kisiIndirim + "%" : "") +
    (kisiIndirim && (mesafeArtis||vardiyaArtis) ? " | " : "") +
    (mesafeArtis  ? "Mesafe: +" + mesafeArtis + "%" : "") +
    (mesafeArtis && vardiyaArtis ? " | " : "") +
    (vardiyaArtis ? "Vardiya: +" + vardiyaArtis + "%" : "") ||
    "Değişiklik yok";

  sonucDiv.style.display = "block";
  uygulaBtn.disabled = false;
}

function mobilTopluUygula() {
  var oran = window._mfp_oran;
  if (oran === undefined) return;
  var rows = document.querySelectorAll("#tblBody tr");
  rows.forEach(function(tr) {
    var grp = tr.querySelector(".r_grup");
    if (grp && grp.value === "MOBIL") {
      var birimInp = tr.querySelector(".r_birim");
      if (birimInp) {
        var katalogFiyat = Number(birimInp.getAttribute("data-katalog") || birimInp.value || 0);
        // İlk uygulamada katalog fiyatını sakla
        if (!birimInp.getAttribute("data-katalog")) {
          birimInp.setAttribute("data-katalog", birimInp.value);
          katalogFiyat = Number(birimInp.value || 0);
        }
        var yeniFiyat = Math.round(katalogFiyat * (1 + oran / 100));
        birimInp.value = yeniFiyat;
        birimInp.dispatchEvent(new Event("input"));
      }
    }
  });
  var pop = document.getElementById("mobilFiyatPopup");
  if (pop) pop.remove();
}
