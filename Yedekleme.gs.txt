function tumDosyalariYedekle() {
  var token = PropertiesService.getScriptProperties().getProperty('GITHUB_TOKEN');
  var repo = 'ulasozbas/AvrupaBirCRM';
  var projectId = ScriptApp.getScriptId();
  
  // Tüm proje dosyalarını çek
  var url = 'https://script.googleapis.com/v1/projects/' + projectId + '/content';
  var response = UrlFetchApp.fetch(url, {
    headers: { 'Authorization': 'Bearer ' + ScriptApp.getOAuthToken() },
    muteHttpExceptions: true
  });
  
  var content = JSON.parse(response.getContentText());
  var files = content.files || [];
  
  var basarili = 0;
  var hatali = [];
  
  files.forEach(function(file) {
    if (!file.source) return;
    
    // Dosya adı: config.gs.txt, Index.html.txt gibi
    var ext = file.type === 'HTML' ? '.html.txt' : '.gs.txt';
    var fileName = file.name + ext;
    var icerik = file.source;
    
    // GitHub'da mevcut SHA'yı al
    var apiUrl = 'https://api.github.com/repos/' + repo + '/contents/' + encodeURIComponent(fileName);
    var sha = '';
    try {
      var mevcut = UrlFetchApp.fetch(apiUrl, {
        headers: {
          'Authorization': 'token ' + token,
          'Accept': 'application/vnd.github.v3+json'
        },
        muteHttpExceptions: true
      });
      if (mevcut.getResponseCode() === 200) {
        sha = JSON.parse(mevcut.getContentText()).sha || '';
      }
    } catch(e) {}
    
    // UTF-8 base64 encode
    var encoded = Utilities.base64Encode(
      Utilities.newBlob(icerik).setContentType('text/plain; charset=utf-8').getBytes()
    );
    
    var body = {
      message: 'Yedek: ' + fileName + ' — ' + Utilities.formatDate(new Date(), 'Europe/Istanbul', 'dd.MM.yyyy HH:mm'),
      content: encoded
    };
    if (sha) body.sha = sha;
    
    try {
      var result = UrlFetchApp.fetch(apiUrl, {
        method: 'PUT',
        headers: {
          'Authorization': 'token ' + token,
          'Accept': 'application/vnd.github.v3+json',
          'Content-Type': 'application/json'
        },
        payload: JSON.stringify(body),
        muteHttpExceptions: true
      });
      var code = result.getResponseCode();
      if (code === 200 || code === 201) {
        basarili++;
        Logger.log('✅ ' + fileName);
      } else {
        hatali.push(fileName + ' (HTTP ' + code + ')');
        Logger.log('❌ ' + fileName + ': ' + result.getContentText().substring(0, 200));
      }
    } catch(e) {
      hatali.push(fileName + ': ' + e.message);
    }
    
    // Rate limit için kısa bekleme
    Utilities.sleep(300);
  });
  
  var mesaj = '✅ ' + basarili + ' dosya yedeklendi.';
  if (hatali.length) mesaj += '\n\n❌ Hata (' + hatali.length + '):\n' + hatali.join('\n');
  Logger.log(mesaj);
  SpreadsheetApp.getUi().alert(mesaj);
}
