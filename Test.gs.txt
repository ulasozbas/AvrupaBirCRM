function firmalariArsivdenGeriAl() {
  var ss = SpreadsheetApp.openById(CRM_SHEET_ID);
  var arsiv = ss.getSheetByName("FIRMALAR_ARSIV_2025");
  var hedef = ss.getSheetByName("FIRMALAR");
  if (!arsiv || !hedef) { Logger.log("Sheet bulunamadı"); return; }
  
  var arsivData = arsiv.getDataRange().getValues();
  if (arsivData.length < 2) { Logger.log("Arşivde veri yok"); return; }
  
  var mevcutData = hedef.getDataRange().getValues();
  var mevcutIdler = {};
  for (var i = 1; i < mevcutData.length; i++) {
    mevcutIdler[String(mevcutData[i][0] || "").trim()] = true;
  }
  
  var eklenen = 0;
  for (var j = 1; j < arsivData.length; j++) {
    var id = String(arsivData[j][0] || "").trim();
    if (!id || mevcutIdler[id]) continue;
    hedef.appendRow(arsivData[j]);
    eklenen++;
  }
  
  Logger.log("Tamamlandı — " + eklenen + " firma arşivden geri alındı.");
}
