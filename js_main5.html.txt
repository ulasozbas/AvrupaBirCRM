function isgGunlukGorevleriYukle() {
  var email = String(window.currentUserEmail || window.aktifKullanici || "").toLowerCase();
  var bugun = new Date().toISOString().split("T")[0];
  if (!email) return;

  var mudurler = [
    "dilara.bayrakdar@avrupabir.com.tr",
    "ulas.ozbas@avrupabir.com.tr",
    "yasemin.orbay@avrupabir.com.tr",
    "gulbahar.gultekin@avrupabir.com.tr"
  ];
  var isMudur = mudurler.indexOf(email) > -1;
  console.log("EMAIL:", email, "MUDUR MU:", isMudur);

  var mudurKutu = document.getElementById("isgTumGunlukPlanlarKutu");
  if (mudurKutu) mudurKutu.style.display = isMudur ? "block" : "none";

  // 1 — Kendi görevleri
  google.script.run
  .withSuccessHandler(function(res) {
    var list = Array.isArray(res) ? res : (res && res.liste ? res.liste : []);
    console.log("KENDİ PLANLAR:", list.length, list);
    var kutu = document.getElementById("isgGunlukGorevlerim");
    if (!kutu) return;
    var benimPlanlar = list.filter(function(p) {
      return String(p.personelEmail||"").toLowerCase() === email;
    });
    if (!benimPlanlar.length) {
      kutu.innerHTML = "<div style='background:#f8fafc; border:1px solid #e2e8f0; border-radius:10px; padding:20px; text-align:center; color:#64748b; margin-bottom:16px;'>Bugün için atanmış göreviniz yok.<\/div>";
    } else {
      kutu.innerHTML = renderGorevler(benimPlanlar);
    }
  })
  .api_isgGunlukPlanGetir(email, bugun);

  // 2 — Müdürse tüm planlar
  if (isMudur) {
    console.log("TUM ÇAĞRISI YAPILIYOR");
    google.script.run
    .withFailureHandler(function(err) {
      console.log("TUM HATA:", err.message);
    })
    .withSuccessHandler(function(tumListe) {
      console.log("TUM LISTE:", tumListe ? tumListe.length : 0, tumListe);
      var tumKutu = document.getElementById("isgTumGunlukPlanlar");
      if (!tumKutu) { console.log("isgTumGunlukPlanlar DIV BULUNAMADI!"); return; }
      if (!tumListe || !tumListe.length) {
        tumKutu.innerHTML = "<div style='text-align:center; padding:20px; color:#64748b;'>Bugün için plan yok.<\/div>";
      } else {
        tumKutu.innerHTML = renderGorevler(tumListe, true);
      }
    })
    .api_isgGunlukPlanGetir("TUM", bugun);
  }
}

function isgAtamaListeYukle() {
  var div = document.getElementById('isgAtamaListesi');
  if (div) div.innerHTML = '<div style="text-align:center;padding:20px;color:#64748b;">Yükleniyor...<\/div>';
  google.script.run
    .withSuccessHandler(function(d) {
      var div = document.getElementById('isgAtamaListesi');
      var badge = document.getElementById('isgAtamaBadge');
      if (!d || !d.ok || !d.liste || !d.liste.length) {
        if (div) div.innerHTML = '<div style="text-align:center;padding:20px;color:#10b981;font-weight:700;">✅ Bekleyen atama yok.<\/div>';
        if (badge) { badge.textContent = '0'; badge.style.background = '#10b981'; }
        return;
      }
      var bekleyen = d.liste.filter(function(a) { return a.durum === 'BEKLEMEDE'; });
      if (badge) { badge.textContent = bekleyen.length; badge.style.background = bekleyen.length > 0 ? '#ef4444' : '#10b981'; }
      var html = '';
      d.liste.forEach(function(a) {
        var durumRenk = a.durum === 'BEKLEMEDE' ? '#fef2f2' : '#f0fdf4';
        var durumTxt = a.durum === 'BEKLEMEDE' ? '🔴 Beklemede' : '✅ Atandı';
        html += '<div style="background:' + durumRenk + ';border:1px solid #e2e8f0;border-radius:12px;padding:16px;">'
          + '<div style="display:flex;justify-content:space-between;align-items:center;margin-bottom:8px;">'
          + '<div style="font-weight:900;font-size:14px;color:#0f172a;">' + (a.firmaAdi || '-') + '<\/div>'
          + '<span style="font-size:11px;font-weight:700;">' + durumTxt + '<\/span></div>'
          + '<div style="font-size:12px;color:#64748b;margin-bottom:4px;">Teklif: ' + (a.teklifNo || '-') + ' | Tarih: ' + (a.tarih || '-') + '<\/div>';
        if (a.durum !== 'BEKLEMEDE') {
          html += '<div style="font-size:12px;color:#065f46;margin-top:6px;">👤 ' + (a.atananPersonel || '-') + ' | 🕐 ' + (a.atananDakika || '-') + ' dk | 📋 ' + (a.hizmetler || '-') + '<\/div>';
        } else {
          html += '<button onclick="atamaModalAc(\'' + a.id + '\',' + a.satir + ',\'' + (a.firmaAdi || '').replace(/'/g, "\\'") + '\',\'' + (a.firmaId || '') + '\')" style="margin-top:8px;background:#065f46;color:#fff;border:none;padding:8px 18px;border-radius:8px;font-weight:700;cursor:pointer;font-size:12px;">🦺 Atama Yap</button>';
        }
        html += '<\/div>';
      });
      if (div) div.innerHTML = html;
    })
    .withFailureHandler(function() { var d = document.getElementById('isgAtamaListesi'); if (d) d.innerHTML = 'Hata.'; })
    .api_bekleyenAtamalar(window.currentUserEmail || "");
}

function atamaModalAc(id, satir, firmaAdi, firmaId) {
  document.getElementById('atamaSatir').value = satir;
  document.getElementById('atamaFirmaId').value = firmaId;
  document.getElementById('atamaFirmaGoster').textContent = '🏢 ' + firmaAdi;
  document.getElementById('atamaCalisan').value = '';
  document.getElementById('atamaHizIGU').checked = false;
  document.getElementById('atamaHizHekim').checked = false;
  document.getElementById('atamaHizDSP').checked = false;
  document.getElementById('atamaNotlar').value = '';
  document.getElementById('atamaPersonelAlanlari').innerHTML = '<div style="color:#94a3b8;font-size:12px;padding:8px;">⏳ Mevcut atamalar yükleniyor...<\/div>';
  atamaPersonelVerisiYukle();
  document.getElementById('atamaModal').style.display = 'flex';

  // Mevcut atamaları yükle
  google.script.run
    .withSuccessHandler(function(r) {
      if (!r || !r.ok || !r.atama) return;
      var a = r.atama;
      if (a.calisan) document.getElementById('atamaCalisan').value = a.calisan;
      if (a.notlar) document.getElementById('atamaNotlar').value = a.notlar;
      if (a.sozlesmeTarihi) {
        var stEl = document.getElementById('atamaSozlesmeTarihi');
        if (stEl) { stEl.value = a.sozlesmeTarihi; stEl.disabled = true; }
      }
      // Hizmetleri işaretle
      var hiz = String(a.hizmetler || '').split(',');
      var per = String(a.personel || '').split(',');
      var dk = String(a.dakika || '').split(',');
      hiz.forEach(function(h, i) {
        h = h.trim();
        if (h === 'IGU') document.getElementById('atamaHizIGU').checked = true;
        if (h === 'Hekim') document.getElementById('atamaHizHekim').checked = true;
        if (h === 'DSP') document.getElementById('atamaHizDSP').checked = true;
      });
      atamaHizmetDegisti();
      // Personel ve dakikaları doldur (dropdown'lar oluştuktan sonra)
      setTimeout(function() {
        hiz.forEach(function(h, i) {
          h = h.trim();
          var pEl = document.getElementById('atamaPer' + h);
          var dEl = document.getElementById('atamaDk' + h);
          if (pEl && per[i]) pEl.value = per[i].trim();
          if (dEl && dk[i]) dEl.value = dk[i].trim();
        });
      }, 500);
    })
    .withFailureHandler(function() {})
    .api_mevcutAtamaGetir(firmaId);
}

function atamaKaydetGonder() {
  var btn = document.getElementById('btnAtamaKaydet');
  btn.disabled = true; btn.textContent = '⏳ Kaydediliyor...';
  var hizmetler = [];
  var personeller = [];
  var dakikalar = [];
  if (document.getElementById('atamaHizIGU').checked) {
    hizmetler.push('IGU');
    personeller.push((document.getElementById('atamaPerIGU') || {}).value || '');
    dakikalar.push((document.getElementById('atamaDkIGU') || {}).value || '0');
  }
  if (document.getElementById('atamaHizHekim').checked) {
    hizmetler.push('Hekim');
    personeller.push((document.getElementById('atamaPerHekim') || {}).value || '');
    dakikalar.push((document.getElementById('atamaDkHekim') || {}).value || '0');
  }
  if (document.getElementById('atamaHizDSP').checked) {
    hizmetler.push('DSP');
    personeller.push((document.getElementById('atamaPerDSP') || {}).value || '');
    dakikalar.push((document.getElementById('atamaDkDSP') || {}).value || '0');
  }

  var payload = {
    satir: document.getElementById('atamaSatir').value,
    firmaId: document.getElementById('atamaFirmaId').value,
    firmaAdi: document.getElementById('atamaFirmaGoster').textContent.replace('🏢 ', ''),
    calisan: document.getElementById('atamaCalisan').value,
    hizmetler: hizmetler.join(','),
    personel: personeller.join(', '),
    dakika: dakikalar.join(','),
    notlar: document.getElementById('atamaNotlar').value,
    atayanEmail: window.currentUserEmail || ''
  };

  google.script.run
    .withSuccessHandler(function(d) {
      btn.disabled = false; btn.textContent = '✅ ATAMAYI KAYDET';
      if (d && d.ok) {
        document.getElementById('atamaModal').style.display = 'none';
        if (typeof notify === 'function') notify('✅ Atama kaydedildi!', 'success');
        isgAtamaListeYukle();
      } else {
        alert('❌ Hata: ' + (d ? d.msg : 'Bilinmiyor'));
      }
    })
    .withFailureHandler(function(e) {
      btn.disabled = false; btn.textContent = '✅ ATAMAYI KAYDET';
      alert('❌ Bağlantı hatası: ' + e.message);
    })
    .api_atamaKaydet(payload);
}

function renderGorevler(list, tumPersonel) {
  var html = '<div style="display:flex; flex-direction:column; gap:8px;">';
  list.forEach(function(p) {
    if (p.durum === "TAMAMLANDI") return;
    var isIptal = String(p.durum||"").indexOf("IPTAL") > -1 || String(p.durum||"").indexOf("\u0130PTAL") > -1;
    var isAraGorev = String(p.firmaId||"").indexOf("GOREV_") > -1 || String(p.hizmetTipi||"").indexOf("Ara G") > -1;
    var pid = String(p.planId||"").replace(/['"]/g,"");
    var firmaAdi = String(p.firmaAdi||"").replace(/['"<>]/g,"");
    var ustCizgi = isIptal ? "text-decoration:line-through;" : "";
    html += '<div style="background:white; border-radius:8px; padding:12px; display:flex; justify-content:space-between; align-items:center; flex-wrap:wrap; gap:8px; border-left:4px solid ' + (isIptal ? '#ef4444' : (isAraGorev ? '#8b5cf6' : '#3b82f6')) + ';">' +
      '<div style="' + ustCizgi + '">' +
      '<div style="font-weight:bold; color:#1e293b; font-size:14px;">' + firmaAdi + '<\/div>' +
      (tumPersonel ? '<div style="font-size:11px; color:#7c3aed; font-weight:bold;">👤 ' + String(p.personelEmail||"").split("@")[0].toUpperCase() + '<\/div>' : '') +
      '<div style="font-size:12px; color:#64748b; margin-top:4px;">' + String(p.hizmetTipi||"") + ' | <b>' + String(p.planlananDk||0) + ' dk<\/b>' + (p.planlananSaat ? ' | ' + p.planlananSaat : '') + '<\/div>' +
      '<\/div><div style="display:flex; gap:6px;">';
    if (isIptal) {
      html += '<span style="color:#ef4444; font-weight:bold; font-size:12px;">IPTAL<\/span>';
    } else if (p.durum === "PLANLANDI" || p.durum === "BEKLIYOR") {
      html += '<button onclick="isgGirisYap(\'' + pid + '\')" style="background:#059669; color:white; border:none; padding:8px 14px; border-radius:6px; cursor:pointer; font-weight:bold; font-size:12px;">' + (isAraGorev ? 'BASLA' : 'TESISE GIR') + '<\/button>';

window.isgKendiRotamiCiz = function() {
    var email = window.currentUserEmail || window.aktifKullanici || "";

function isgGirisYap(planId) {
  if (!navigator.geolocation) { 
    alert("Tarayıcınız konum özelliğini desteklemiyor!"); 
    return; 
  }

  var input = document.createElement("input");
  input.type = "file";
  input.accept = "image/*";
  input.capture = "environment";
  input.style.display = "none";
  document.body.appendChild(input);

  input.onchange = function() {
    if (input.files && input.files[0]) {
      var reader = new FileReader();
      reader.onload = function(e) { devamEt(e.target.result); };
      reader.readAsDataURL(input.files[0]);
    } else {
      devamEt("");
    }
    document.body.removeChild(input);
  };

  setTimeout(function() {
    if (document.body.contains(input)) {
      document.body.removeChild(input);
      devamEt("");
    }
  }, 60000);

  input.click();

  function devamEt(fotoBase64) {
    notify("📍 Konum aranıyor...", "info");
    navigator.geolocation.getCurrentPosition(
      function(pos) {
        var userLat = pos.coords.latitude;
        var userLng = pos.coords.longitude;
        var konum = userLat + "," + userLng;
        var saat = new Date().toLocaleTimeString("tr-TR");
        google.script.run
          .withSuccessHandler(function(firmaKonum) {
            var gpsUzak = false;
            var mesafe = 0;
            if (firmaKonum && firmaKonum.enlem && firmaKonum.boylam) {
              var R = 6371000;
              var dLat = (firmaKonum.enlem - userLat) * Math.PI / 180;
              var dLng = (firmaKonum.boylam - userLng) * Math.PI / 180;
              var a = Math.sin(dLat/2) * Math.sin(dLat/2) +
                      Math.cos(userLat * Math.PI/180) * Math.cos(firmaKonum.enlem * Math.PI/180) *
                      Math.sin(dLng/2) * Math.sin(dLng/2);
              mesafe = Math.round(R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a)));
              gpsUzak = mesafe > 150;
              if (gpsUzak) notify("⚠️ Firmaya " + mesafe + " metre uzaktasınız! Giriş yine de kaydedilecek.", "warning");
            }
            google.script.run
              .withSuccessHandler(function(res) {
                if (res && res.ok) {
                  notify("✅ Tesis girişi kaydedildi! Saat: " + saat, "success");
                  isgGunlukGorevleriYukle();
                } else {
                  alert("❌ Hata: " + (res ? res.msg : "Bilinmiyor"));
                }
              })
              .withFailureHandler(function(err) { alert("❌ Bağlantı Hatası: " + err.message); })
              .api_isgGirisKaydet({ planId: planId, konum: konum, saat: saat, email: window.currentUserEmail, foto: fotoBase64, gpsUzak: gpsUzak, mesafe: mesafe });
          })
          .withFailureHandler(function() {
            google.script.run
              .withSuccessHandler(function(res) {
                if (res && res.ok) {
                  notify("✅ Tesis girişi kaydedildi! Saat: " + saat, "success");
                  isgGunlukGorevleriYukle();
                } else {
                  alert("❌ Hata: " + (res ? res.msg : "Bilinmiyor"));
                }
              })
              .api_isgGirisKaydet({ planId: planId, konum: konum, saat: saat, email: window.currentUserEmail, foto: fotoBase64 });
          })
          .api_firmaKonumGetir(planId);
      },
      function(error) {
        var msg = "Konum alınamadı!";
        if (error.code === 1) msg = "Konum izni reddedildi!";
        else if (error.code === 2) msg = "GPS sinyali bulunamadı.";
        else if (error.code === 3) msg = "Konum bulma süresi doldu.";
        alert("❌ " + msg);
      },
      { enableHighAccuracy: true, timeout: 15000, maximumAge: 0 }
    );
  }
}

        if (yukleniyor) yukleniyor.style.display = "none";
        
        if (!res || !res.ok || !res.firmalar) {
          if (listeKutu) listeKutu.innerHTML = "<div style='color:red; padding:10px;'>Firmalar yüklenemedi.<\/div>";
          return;
        }

        var firmalar = res.firmalar;
        var tamamlanan = 0;
        var htmlList = "";
        var bounds = [];
        
        var hedefFirmaId = document.getElementById("isgPlanFirmaId") ? document.getElementById("isgPlanFirmaId").value : "";

        firmalar.forEach(function(f) {
          if (!f.lat || !f.lng) return; 
          
          var isTarget = (String(f.firmaId) === String(hedefFirmaId));
          var mIcon = isTarget ? redIcon : greenIcon;
          
          var marker = L.marker([f.lat, f.lng], {icon: mIcon}).addTo(modalMap);
          
          if (isTarget) {
             marker.bindPopup("<div style='text-align:center;'><b style='color:#dc2626; font-size:14px;'>📍 HEDEF FİRMA<\/b><br><b>" + window.esc(f.firmaAdi) + "<\/b><\/div>").openPopup();
             modalMap.setView([f.lat, f.lng], 13);
          } else {
             marker.bindPopup(
                "<div style='min-width:180px;'>" +
                "<b style='color:#065f46; font-size:13px;'>" + window.esc(f.firmaAdi) + "<\/b><br>" +
                "<span style='font-size:11px; color:#64748b;'>" + window.esc(f.adres) + "<\/span><br><br>" +
                "<button onclick=\"window.isgEkstraFirmaEkle('" + f.firmaId + "','" + window.esc(f.firmaAdi).replace(/'/g,"") + "')\" " +

                "<\/div>"
             );
          }
          
          markers[f.firmaId] = marker;
          bounds.push([f.lat, f.lng]);
          tamamlanan++;

          if(!isTarget) {
              htmlList += "<div onclick=\"var m=modalMap_markers['" + f.firmaId + "']; if(m){modalMap_instance.setView(m.getLatLng(),15); m.openPopup();}\" " +
              "style='padding:8px; border:1px solid #cbd5e1; border-radius:6px; margin-bottom:4px; cursor:pointer; background:white;' " +
              "onmouseover=\"this.style.borderColor='#065f46'\" onmouseout=\"this.style.borderColor='#cbd5e1'\">" +
              "<div style='font-size:11px; font-weight:bold; color:#1e293b;'>" + window.esc(f.firmaAdi) + "<\/div>" +
              "<div style='font-size:10px; color:#64748b;'>" + window.esc(f.adres) + "<\/div>" +
              "<\/div>";
          }
        });

        window.modalMap_instance = modalMap;
        window.modalMap_markers = markers;

        if (listeKutu) listeKutu.innerHTML = htmlList;
        if (pinSayaci) pinSayaci.textContent = tamamlanan + " firma haritada";

        if(bounds.length > 0 && !markers[hedefFirmaId]) {
            modalMap.fitBounds(bounds, {padding: [30,30]});
        }

        setTimeout(function() { modalMap.invalidateSize(); }, 300);

      })
      .withFailureHandler(function(err) {
        if (yukleniyor) yukleniyor.style.display = "none";
        if (pinSayaci) pinSayaci.textContent = "Hata oluştu";
      })
      .api_isgHaritaVerisiGetir(); 
  }

  if (typeof L === "undefined") {
    var css = document.createElement("link");
    css.rel = "stylesheet";
    css.href = "https://unpkg.com/leaflet@1.9.4/dist/leaflet.css";
    document.head.appendChild(css);

    var js = document.createElement("script");
    js.src = "https://unpkg.com/leaflet@1.9.4/dist/leaflet.js";
    js.onload = function() { haritayiBaslat(); };
    js.onerror = function() { alert("Harita yuklenemedi!"); };
    document.head.appendChild(js);
  } else {
    haritayiBaslat();
  }
};

window.isgPlanKaydet = function() {
  var tarih  = (document.getElementById("isgPlanTarihModal")||{}).value || "";
  var firmaId  = (document.getElementById("isgPlanFirmaId")||{}).value || "";
  var firmaAdi = (document.getElementById("isgPlanFirmaAdi")||{}).value || "";
  var personel = (document.getElementById("isgPlanPersonel")||{}).value || "";
  var hizmet  = (document.getElementById("isgPlanHizmet")||{}).value || "";
  var sure  = parseInt((document.getElementById("isgPlanSure")||{}).value) || 0;
  var saat  = (document.getElementById("isgPlanSaat")||{}).value || "";
  var not_  = (document.getElementById("isgPlanNot")||{}).value || "";

  var planlar = [];

  // 1. Ana firma kutusu doluysa onu plana dahil et
  if (firmaId && sure > 0) {
      planlar.push({
          tarih: tarih, firmaId: firmaId, firmaAdi: firmaAdi,
          personelEmail: personel, hizmetTipi: hizmet,
          planlananDk: sure, notlar: not_,
          planlananSaat: saat,
      });
  }

  // 2. Haritadan eklenen ekstra firmaları al
  var ekstraRows = document.querySelectorAll(".ekstraFirmaRow");
  for (var i = 0; i < ekstraRows.length; i++) {
      var row = ekstraRows[i];
      var eFirmaId  = row.getAttribute("data-id");
      var eFirmaAdi = row.getAttribute("data-ad");
      var eSure  = parseInt(row.querySelector(".ekstraSure").value) || 0;
      var eNot  = row.querySelector(".ekstraNot").value || "";
      
      if (!eFirmaId || eSure <= 0) continue;
      
      planlar.push({
          tarih: tarih, firmaId: eFirmaId, firmaAdi: eFirmaAdi,
          personelEmail: personel, hizmetTipi: hizmet,
          planlananDk: eSure, notlar: eNot,
          olusturanEmail: window.currentUserEmail || ""
      });
  }

  // Temel doğrulamalar (FirmaID kontrolünü kaldırdık, çünkü planlar dizisini kontrol ediyoruz)
  if (!tarih || !personel || !hizmet) {
      alert("Tarih, gidecek personel ve hizmet tipi zorunludur!"); return;
  }

  if (planlar.length === 0) {
      alert("Lütfen en az bir firma için geçerli bir süre (dk) girin!"); return;
  }

  var mesaj = document.getElementById("isgPlanMesaj");
  if (mesaj) mesaj.textContent = "Sisteme İşleniyor... (" + planlar.length + " plan)";

  google.script.run
  .withSuccessHandler(function(res) {
      if (res && res.ok) {
          if (mesaj) mesaj.textContent = "✅ " + planlar.length + " plan başarıyla kaydedildi!";
          // WhatsApp — her firma için ayrı link
          planlar.forEach(function(plan) {
            if (!plan.firmaId || plan.firmaId.indexOf("GOREV_") > -1) return;
            var waMesaj = "Sayın " + plan.firmaAdi + ",\n\n" +
              plan.tarih + " tarihinde " + plan.hizmetTipi + " hizmetiniz için ziyaret planlanmıştır.\n" +
              "Süre: " + plan.planlananDk + " dk\n\n" +
              "Bilginize sunarız.\nAvrupaİSG OSGB\n0549 483 83 82";
            google.script.run.withSuccessHandler(function(wa) {
              if (wa && wa.ok && wa.url) {
                if (confirm("📱 " + plan.firmaAdi + " firmasına WhatsApp mesajı gönderilsin mi?")) {
                  window.open(wa.url, "_blank");
                }
              }
            }).api_waLink({ firmaId: plan.firmaId, mesaj: waMesaj });
          });
          setTimeout(function() {
              var m = document.getElementById("isgPlanModal");
              if (m) m.remove();
              if(typeof isgPlanlariYukle === "function") isgPlanlariYukle();
          }, 1500);
      } else {
          if (mesaj) mesaj.textContent = "❌ Hata: " + (res ? res.msg : "?");
      }
  })
  .withFailureHandler(function(err) {
      if (mesaj) mesaj.textContent = "❌ Bağlantı Hatası: " + err.message;
  })
  .api_isgCokluPlanKaydet(planlar);
};

function isgGunlukRotaAc() {
  var tarihEl = document.getElementById("isgPlanTarih");
  var tarih = tarihEl ? tarihEl.value : new Date().toISOString().split("T")[0];
  if (!tarih) { alert("Önce tarih seçin!"); return; }

  google.script.run
  .withSuccessHandler(function(list) {
  if (!list || !list.length) {
  notify("Bu tarih için plan bulunamadı.", "info");
  return;
  }

  var adresler = list.map(function(p) {
  return encodeURIComponent(String(p.firmaAdi||"") + " İstanbul");
  });

  if (adresler.length === 1) {
  window.open("https://www.google.com/maps/search/?api=1&query=" + adresler[0], "_blank");
  return;
  }

  var destination = adresler[adresler.length - 1];
  var waypoints = adresler.slice(0, -1).join("|");
  var url = "https://www.google.com/maps/dir/?api=1&destination=" + destination + "&waypoints=" + waypoints + "&travelmode=driving";
  window.open(url, "_blank");
  })
  .api_isgPlanListeGetir(tarih, window.currentUserEmail, window.userRole);
}

function meditekImportEt() {
  var file = document.getElementById("meditekImportFile").files[0];
  if (!file) return;

  notify("📥 İSG-KATİP Exceli okunuyor...", "info");

  var reader = new FileReader();
  reader.onload = function(e) {
  try {
  var data = new Uint8Array(e.target.result);
  var workbook = XLSX.read(data, { type: "array" });
  var sheet = workbook.Sheets[workbook.SheetNames[0]];
  var rows = XLSX.utils.sheet_to_json(sheet, { header: 1 });

  if (rows.length < 2) { notify("❌ Excel boş veya hatalı!", "error"); return; }

  var basliklar = rows[0];

  var bul = function(arananKelimeler) {
  for (var x = 0; x < basliklar.length; x++) {
  var baslikTemiz = String(basliklar[x]).toLowerCase()
  .replace(/i̇/g,"i").replace(/ı/g,"i").replace(/İ/g,"i")
  .replace(/ş/g,"s").replace(/ç/g,"c").replace(/ö/g,"o")
  .replace(/ü/g,"u").replace(/ğ/g,"g").replace(/\s/g,"");
  for (var y = 0; y < arananKelimeler.length; y++) {
  if (baslikTemiz.indexOf(arananKelimeler[y]) > -1) return x;
  }
  }
  return -1;
  };

  var cxFirma  = bul(["hizmetalanisyeriunvani"]);
  var cxSgk  = bul(["hizmetalanisyerisgk"]);
  var cxCalisan= bul(["calisansayisi"]);
  var cxTehlike= bul(["tehlikesinifi"]);
  var cxBas  = bul(["baslangictarihi"]);
  var cxBit  = bul(["bitistarihi"]);
  var cxDurum  = bul(["sozlesmestatusu"]);
  var cxPerAd  = bul(["gorevlendirilenkisiad"]);
  var cxSertif = bul(["sertifikatipi"]);
  var cxSure  = bul(["calismasuresi"]);
  var cxSozId  = bul(["sozlesmeid"]);
  var cxAdres  = bul(["firmaadresi","adres"]);

  if (cxFirma === -1) { 
  notify("❌ Excelde 'Hizmet Alan İşyeri Unvanı' sütunu bulunamadı!", "error"); 
  return; 
  }

  var hamKayitlar = [];
  for (var i = 1; i < rows.length; i++) {
  var r = rows[i];
  if (!r[cxFirma]) continue; 
  var durum = cxDurum > -1 ? String(r[cxDurum]||"").toLowerCase() : "";
  if (durum.indexOf("devam") === -1 && durum.indexOf("onaylandi") === -1 && durum.indexOf("onaylandı") === -1) continue;

  var adresHam = cxAdres > -1 ? String(r[cxAdres]||"").trim() : "";
  hamKayitlar.push({
  sgkSicilNo:  cxSgk > -1 ? String(r[cxSgk]||"") : "",
  firmaAdi:  String(r[cxFirma]||""),
  calisanSayisi:  cxCalisan > -1 ? parseInt(r[cxCalisan])||0 : 0,
  tehlikeSinifi:  cxTehlike > -1 ? String(r[cxTehlike]||"") : "",
  sozlesmeBas:  (cxBas > -1 && r[cxBas]) ? new Date(r[cxBas]).toISOString().split("T")[0] : "",
  sozlesmeBit:  (cxBit > -1 && r[cxBit]) ? new Date(r[cxBit]).toISOString().split("T")[0] : "",
  atamaliPersonel: cxPerAd > -1 ? String(r[cxPerAd]||"") : "",
  sertifikaTipi:  cxSertif > -1 ? String(r[cxSertif]||"") : "",
  calismaSuresi:  cxSure > -1 ? parseInt(r[cxSure])||0 : 0,
  sozlesmeId:  cxSozId > -1 ? String(r[cxSozId]||"") : "",
  adres:  adresHam
  });
  }

  hamKayitlar.forEach(function(k) {
  var key = k.sgkSicilNo && k.sgkSicilNo !== "" ? k.sgkSicilNo : k.firmaAdi;
  if (!sgkMap[key]) {
  sgkMap[key] = k;
  } else {
  var m = sgkMap[key];
  if (!m.adres && k.adres) m.adres = k.adres;
  if (k.calisanSayisi > m.calisanSayisi) m.calisanSayisi = k.calisanSayisi;
  if (k.sozlesmeBit > m.sozlesmeBit) m.sozlesmeBit = k.sozlesmeBit;
  }
  });

  progressDiv.innerHTML = '<div style="background:#7c3aed; height:5px; animation:progressAnim 3s ease-in-out infinite;"><\/div>' +
  '<div style="background:rgba(124,58,237,0.9); color:white; padding:12px 20px; font-weight:bold; font-size:13px; text-align:center;">📥 ' + kayitlar.length + ' kayıt İSG-KATİP listesinden güncelleniyor... Lütfen bekleyin ⏳<\/div>';
  document.body.appendChild(progressDiv);

  google.script.run
  .withSuccessHandler(function(res) {
  var pg = document.getElementById("meditekProgress");
  if (pg) pg.remove();
  if (res && res.ok) {
  notify("✅ " + res.eklenen + " yeni, " + res.guncellenen + " güncellendi!", "success");
  document.getElementById("meditekImportFile").value = "";
  if(typeof isgFirmalariYukle === "function") isgFirmalariYukle();
  } else {
  notify("❌ " + (res ? res.msg : "Hata"), "error");
  }
  })
  .withFailureHandler(function(err) {
  var pg = document.getElementById("meditekProgress");
  if (pg) pg.remove();
  notify("❌ " + err.message, "error");
  })
  .api_meditekImport(kayitlar);

} catch(err) {
  notify("Excel okuma hatasi: " + err.message, "error");
  }
  };

function showPanel(panelId) {
  var allPanels = [
  "panelFirmalar", "panelTeklif", "panelGecmis", "panelRapor", "panelPerf",
  "panelPolice", "panelHarita", "panelFinans", "panelPatronUyari", "panelPersonel",
  "panelISG", "panelYetki", "panelTeklifGecmis", "panelTopluMail",
  "panelDenetim", "panelPerformans", "panelCEO", "panelMobil", "panelBanko", "panelSosyalMedya", "panelROI"
];
  // Tüm panelleri gizle
  allPanels.forEach(function(id) {
    var el = document.getElementById(id);
    if (el) {
      el.style.display = "none";
      el.classList.add("hidden-panel");
    }
  });

  // İsim uyuşmazlıklarını otomatik çözen akıllı yönlendirici
  var target = document.getElementById(panelId);
  if (!target && panelId === "panelGecmis") target = document.getElementById("panelTeklifGecmis");
  if (!target && panelId === "panelTeklifGecmis") target = document.getElementById("panelGecmis");
  if (!target && panelId === "panelPerf") target = document.getElementById("panelPerformans");
  if (!target && panelId === "panelPerformans") target = document.getElementById("panelPerf");
  if (!target && panelId === "panelPolice") target = document.getElementById("panelDenetim");
  if (!target && panelId === "panelDenetim") target = document.getElementById("panelPolice");

// Hedef paneli göster
  if (target) {
    target.style.display = "block";
    target.classList.remove("hidden-panel");
  }

  // panelBanko özel kontrol
  if (panelId === "panelBanko") {
    var bankoEl = document.getElementById("panelBanko");
    if (bankoEl) {
      bankoEl.style.display = "block";
      bankoEl.classList.remove("hidden-panel");
    }
  }

 if (panelId === "panelFinans") {
  muh2Yukle();
}

  window.scrollTo(0, 0);
}
