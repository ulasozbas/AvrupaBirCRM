// GENERIC MENU BUTTON EVENT HANDLER
// Tüm data-panel'li buttonlar için otomatik panel açma
// ═══════════════════════════════════════════════════════════════════
document.addEventListener('DOMContentLoaded', function() {
  var menuButtons = document.querySelectorAll('.menuBtn[data-panel]');
  menuButtons.forEach(function(btn) {
    if (!btn.onclick) {
      btn.addEventListener('click', function(e) {
        var panelId = this.getAttribute('data-panel');
        if (panelId && typeof showPanel === 'function') {
          showPanel(panelId);
        }
      });
    }
  });
});
