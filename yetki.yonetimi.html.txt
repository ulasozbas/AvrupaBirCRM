<script>
var _roller = {};
var _tanimlar = [];
var _aktifRol = "ADMIN";
var _degisti = false;
var _kullaniciEmail = "";

// ── BAŞLAT ────────────────────────────────────────────────────
window.onload = function() {
  // Login email'i al (GAS ortamında)
  if (typeof google !== 'undefined') {
    google.script.run
      .withSuccessHandler(function(res) {
        if (res && res.email) _kullaniciEmail = res.email;
      })
      .api_oturumBilgisiGetir();
  }
  yukle();
};

function yukle() {
  if (typeof google !== 'undefined') {
    google.script.run
      .withSuccessHandler(function(res) {
        if (!res.ok && !res.roller) {
          toast("Yetki verileri alınamadı: " + (res.msg || ""), "hata");
          return;
        }
        _roller = JSON.parse(JSON.stringify(res.roller));
        _tanimlar = res.tanimlar;
        guncelleSayaclar();
        rolSec("ADMIN");
      })
      .withFailureHandler(function(err) {
        toast("Bağlantı hatası: " + err.message, "hata");
      })
      .api_yetkiRolleriniGetir();
  } else {
    // Geliştirme modu — varsayılan verilerle çalış
    fetch('').catch(function() {});
    toast("GAS ortamı dışında çalışıyor — demo mod", "hata");
  }
}

// ── ROL SEÇ ──────────────────────────────────────────────────
function rolSec(rol) {
  _aktifRol = rol;
  document.querySelectorAll('.rol-btn').forEach(function(b) {
    b.classList.toggle('aktif', b.dataset.rol === rol);
  });
  izinleriBoyaKalan(rol);
}

function izinleriBoyaKalan(rol) {
  var icerik = document.getElementById('icerik');
  var izinler = _roller[rol] || {};
  var adminMi = (rol === "ADMIN");
  var sayac = 0;
  Object.values(izinler).forEach(function(v) { if (v) sayac++; });

  var gruplar = {};
  _tanimlar.forEach(function(t) {
    if (!gruplar[t.grup]) gruplar[t.grup] = [];
    gruplar[t.grup].push(t);
  });

  var grupIkonlari = {
    "Dashboard": "📊", "Firmalar": "🏢", "Teklifler": "📄",
    "Finans": "💰", "Personel/İK": "👤", "ISG": "🦺",
    "Raporlar": "📈", "Sistem": "⚙️"
  };

  var html = '';

  if (adminMi) {
    html += '<div class="admin-uyari">🔒 Admin rolü her zaman tüm izinlere sahiptir — değiştirilemez.</div>';
  }

  html += '<div class="tab-bar"><button class="tab-btn aktif">İzinler</button></div>';

  Object.keys(gruplar).forEach(function(grup) {
    var tanimlar = gruplar[grup];
    var grupAcikSayisi = tanimlar.filter(function(t) { return izinler[t.key] === true; }).length;
    var ikon = grupIkonlari[grup] || "📌";

    html += '<div class="grup-baslik">';
    html += '<h3>' + ikon + ' ' + grup + '</h3>';
    html += '<span class="izin-sayac">' + grupAcikSayisi + '/' + tanimlar.length + ' açık</span>';
    if (!adminMi) {
      html += '<button class="grup-tumunu-sec" onclick="grupTumunuToggle(\'' + grup + '\')">'
            + 'Tümünü seç/kaldır</button>';
    }
    html += '</div>';

    tanimlar.forEach(function(t) {
      var aktif = adminMi ? true : (izinler[t.key] === true);
      var checkedAttr = aktif ? 'checked' : '';
      var disabledAttr = adminMi ? 'disabled' : '';
      var disabledClass = adminMi ? 'disabled' : '';

      html += '<div class="izin-satir" data-grup="' + grup + '">';
      html += '<div class="izin-etiket">' + t.etiket
            + '<span class="izin-key">' + t.key + '</span></div>';
      html += '<label class="toggle-wrap ' + disabledClass + '">';
      html += '<input type="checkbox" ' + checkedAttr + ' ' + disabledAttr
            + ' onchange="izinDegisti(\'' + t.key + '\', this.checked)">';
      html += '<span class="toggle-slider"></span>';
      html += '</label>';
      html += '</div>';
    });
  });

  icerik.innerHTML = html;
}

// ── İZİN DEĞİŞTİ ─────────────────────────────────────────────
function izinDegisti(key, deger) {
  if (!_roller[_aktifRol]) _roller[_aktifRol] = {};
  _roller[_aktifRol][key] = deger;
  _degisti = true;
  document.getElementById('btnKaydet').disabled = false;
  guncelleSayaclar();
}

// ── GRUP TÜMÜNÜ TOGGLE ────────────────────────────────────────
function grupTumunuToggle(grup) {
  var satirlar = document.querySelectorAll('.izin-satir[data-grup="' + grup + '"]');
  var izinler = _roller[_aktifRol] || {};
  var hedefGrup = _tanimlar.filter(function(t) { return t.grup === grup; });
  var tumAcik = hedefGrup.every(function(t) { return izinler[t.key] === true; });
  var yeniDeger = !tumAcik;

  satirlar.forEach(function(satir) {
    var cb = satir.querySelector('input[type=checkbox]');
    if (cb) {
      cb.checked = yeniDeger;
      var key = cb.closest('.izin-satir').querySelector('.izin-key').textContent;
      izinDegisti(key, yeniDeger);
    }
  });
}

// ── SAYAÇLARI GÜNCELLE ────────────────────────────────────────
function guncelleSayaclar() {
  Object.keys(_roller).forEach(function(rol) {
    var izinler = _roller[rol] || {};
    var acik = Object.values(izinler).filter(Boolean).length;
    var el = document.getElementById('sayac-' + rol);
    if (el) el.textContent = acik;
  });
}

// ── KAYDET ───────────────────────────────────────────────────
function kaydet() {
  document.getElementById('btnKaydet').disabled = true;
  if (typeof google !== 'undefined') {
    google.script.run
      .withSuccessHandler(function(res) {
        if (res.ok) {
          toast("✅ Yetki tablosu kaydedildi!", "basarili");
          _degisti = false;
        } else {
          toast("❌ Kayıt hatası: " + res.msg, "hata");
          document.getElementById('btnKaydet').disabled = false;
        }
      })
      .withFailureHandler(function(err) {
        toast("❌ Bağlantı hatası: " + err.message, "hata");
        document.getElementById('btnKaydet').disabled = false;
      })
      .api_yetkiKaydet({ email: _kullaniciEmail, roller: _roller });
  } else {
    toast("GAS ortamı dışında — kayıt yapılamaz", "hata");
    document.getElementById('btnKaydet').disabled = false;
  }
}

// ── VARSAYILANLARA DÖN ────────────────────────────────────────
function varsayilanlariYukle() {
  if (!confirm('Tüm roller varsayılan değerlere döndürülecek. Emin misiniz?')) return;
  yukle();
  _degisti = false;
  document.getElementById('btnKaydet').disabled = true;
  toast("Varsayılan değerler yüklendi", "basarili");
}

// ── KULLANICI LİSTESİ ─────────────────────────────────────────
function kullaniciTabiniGoster() {
  document.querySelectorAll('.rol-btn').forEach(function(b) {
    b.classList.toggle('aktif', b.dataset.rol === '__kullanicilar');
  });
  var icerik = document.getElementById('icerik');
  icerik.innerHTML = '<div class="yukleniyor">Kullanıcılar yükleniyor...</div>';

  if (typeof google !== 'undefined') {
    google.script.run
      .withSuccessHandler(kullanicilariRender)
      .withFailureHandler(function(err) {
        icerik.innerHTML = '<div style="color:#f87171;padding:20px">Hata: ' + err.message + '</div>';
      })
      .api_kullaniciListesiGetir(_kullaniciEmail);
  } else {
    icerik.innerHTML = '<div style="color:#f87171;padding:20px">GAS ortamı dışında kullanıcı listesi görüntülenemez.</div>';
  }
}

function kullanicilariRender(res) {
  if (!res.ok) {
    document.getElementById('icerik').innerHTML =
      '<div style="color:#f87171;padding:20px">❌ ' + res.msg + '</div>';
    return;
  }
  var liste = res.liste || [];
  var html = '<table class="tablo"><thead><tr>'
    + '<th>E-Posta</th><th>Ad Soyad</th><th>Mevcut Rol</th><th>Durum</th><th>İşlem</th>'
    + '</tr></thead><tbody>';

  liste.forEach(function(k) {
    var rolSecenekleri = ['ADMIN','MUDUR','ISG','MUHASEBE','IK','PAZARLAMA','MOBIL','PERSONEL']
      .map(function(r) {
        return '<option value="' + r + '"' + (k.rol === r ? ' selected' : '') + '>' + r + '</option>';
      }).join('');

    html += '<tr>'
      + '<td>' + k.email + '</td>'
      + '<td>' + (k.adSoyad || '—') + '</td>'
      + '<td><select class="rol-secici" onchange="kullaniciRoluGuncelle(\'' + k.email + '\', this.value)">'
      + rolSecenekleri + '</select></td>'
      + '<td>' + (k.aktif ? '<span class="aktif-badge">● Aktif</span>' : '<span class="pasif-badge">● Pasif</span>') + '</td>'
      + '<td><span style="color:#475569;font-size:11px;">' + (k.tel || '') + '</span></td>'
      + '</tr>';
  });

  html += '</tbody></table>';
  document.getElementById('icerik').innerHTML = html;
}

function kullaniciRoluGuncelle(hedefEmail, yeniRol) {
  if (typeof google !== 'undefined') {
    google.script.run
      .withSuccessHandler(function(res) {
        toast(res.ok ? ('✅ ' + res.msg) : ('❌ ' + res.msg), res.ok ? 'basarili' : 'hata');
      })
      .api_kullaniciRoluGuncelle({ adminEmail: _kullaniciEmail, hedefEmail: hedefEmail, yeniRol: yeniRol });
  }
}

// ── TOAST ─────────────────────────────────────────────────────
function toast(mesaj, tip) {
  var el = document.getElementById('toast');
  el.textContent = mesaj;
  el.className = 'toast ' + (tip || 'basarili');
  el.style.display = 'block';
  setTimeout(function() { el.style.display = 'none'; }, 3500);
}
